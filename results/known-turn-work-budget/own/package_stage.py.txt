"""Seal the one complete capped four-path own study before reference audit."""
import gzip
import hashlib
import json
import shutil
from pathlib import Path
from known_turn_work_budget.plan import CANDIDATE,CONTROL,GAMES,LABELS,READOUTS,contract,paths

root=Path(__file__).resolve().parents[3]
folder=Path(__file__).resolve().parent
out=root/'results/known-turn-work-budget/own'
source='2a64f9e0e5cc620d50d43e595ab3e4e1e4b3414a729b5d487048f7a33e410d7a'
head='e3855f7fa579cf78abe53b55ac7df8381895ad2d'
if out.exists() or out.is_symlink():raise FileExistsError('retain earlier own study')
plan=contract(root)
state=json.loads((folder/'job-state.json').read_bytes())
result=json.loads((folder/'raw/result.json').read_bytes())
assert state['status']=='passed' and state['exit_code']==0 and state['cap_seconds']==7200 and state['reserve_within_cap_seconds']==120 and 0<state['observed_wall_seconds']<7200
assert state['source_head']==head and state['source_fingerprint']==source and (folder/'launch.exit').read_text().strip()=='0'
assert result['status']=='completed' and result['stage']=='study' and result['head']==head and result['source_fingerprint']==source
assert result['assignments']==[[0,2],[1,3]] and result['worker_count']==2
assert len(result['workers'])==2 and all(w['exitcode']==0 for w in result['workers'])
for spec in paths():
 saved=json.loads((folder/'raw/study'/f'path{spec.index:02d}/summary.json').read_bytes())
 assert saved['status']=='completed' and saved['path']==dict(index=spec.index,game=spec.game,arm=spec.arm,seed=0)
 assert saved['labels']==list(LABELS) and len(saved['captures'])==6 and saved['recall_violations']==0
 role='wet' if spec.game==GAMES[0] else 'dry'
 arm='control' if spec.arm==CONTROL else 'candidate'
 for capture in saved['captures']:
  meta=capture['metadata'];work=meta['work_label'];readout=meta['readout']
  assert work in LABELS and readout in READOUTS and meta['total_work']>=work
  if work==LABELS[0]:
   assert meta['total_work']==plan['historical_10m_actual_work'][role]
   assert capture['receipt']['policy_sha256']==plan['historical_10m_policy_sha256'][f'{role}_{arm}_{readout}']
for a,b in ((0,1),(2,3)):
 left=json.loads((folder/'raw/study'/f'path{a:02d}/summary.json').read_bytes())
 right=json.loads((folder/'raw/study'/f'path{b:02d}/summary.json').read_bytes())
 for label in LABELS:
  assert {c['metadata']['total_work'] for c in left['captures'] if c['metadata']['work_label']==label}=={c['metadata']['total_work'] for c in right['captures'] if c['metadata']['work_label']==label}
assert not any(path.is_symlink() for path in (folder/'raw').rglob('*'))
shutil.copytree(folder/'raw',out)
for name in ('job-state.json','launch.exit','launch.log','launcher.log','launcher.py','package_stage.py','proof-head-canceled.json','profile-head-canceled.json'):
 shutil.copyfile(folder/name,out/(name+'.txt' if name.endswith('.py') else name))
shutil.copyfile(root/'results/runs/known-turn-work-budget-proof-20261124/source-receipt-head-canceled.json',out/'source-receipt-head-canceled.json')
for label,origin in (
 ('proof-head-canceled',folder),
 ('profile-head-canceled',folder),
 ('source-receipt-head-canceled',root/'results/runs/known-turn-work-budget-proof-20261124'),
):
 with (out/f'{label}.log.gz').open('xb') as stream:
  with gzip.GzipFile(filename='',mode='wb',fileobj=stream,mtime=0) as zipped:
   zipped.write((origin/f'{label}.log').read_bytes())
files=[]
for path in sorted(p for p in out.rglob('*') if p.is_file()):
 blob=path.read_bytes();files.append(dict(file=path.relative_to(out).as_posix(),bytes=len(blob),sha256=hashlib.sha256(blob).hexdigest()))
(out/'manifest.json').write_text(json.dumps(dict(status='own_published',source_fingerprint=source,files=files),sort_keys=True,indent=2)+'\n')
print('own_published',len(files),out)
