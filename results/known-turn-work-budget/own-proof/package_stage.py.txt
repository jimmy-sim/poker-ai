"""Seal the single capped fictional proof before real profiling."""
import hashlib
import json
import shutil
from pathlib import Path

root=Path(__file__).resolve().parents[3]
folder=Path(__file__).resolve().parent
out=root/'results/known-turn-work-budget/own-proof'
source='2a64f9e0e5cc620d50d43e595ab3e4e1e4b3414a729b5d487048f7a33e410d7a'
head='6acd228969fc9c53853f8cabadcc970644c9263d'
if out.exists() or out.is_symlink():raise FileExistsError('retain earlier proof')
state=json.loads((folder/'job-state.json').read_bytes())
result=json.loads((folder/'raw/result.json').read_bytes())
assert state['status']=='passed' and state['exit_code']==0 and state['cap_seconds']==180 and 0<state['observed_wall_seconds']<180
assert state['source_head']==head and state['source_fingerprint']==source
assert result['status']=='completed' and result['stage']=='proof' and result['head']==head and result['source_fingerprint']==source
assert result['fixture']=='fictional signal game' and result['captures_per_arm']==6 and result['final_work']>=720
assert (folder/'launch.exit').read_text().strip()=='0'
assert not any(path.is_symlink() for path in (folder/'raw').rglob('*'))
shutil.copytree(folder/'raw',out)
for name in ('job-state.json','launch.exit','launch.log','launcher.log','launcher.py','package_stage.py'):
 shutil.copyfile(folder/name,out/(name+'.txt' if name.endswith('.py') else name))
files=[]
for path in sorted(p for p in out.rglob('*') if p.is_file()):
 blob=path.read_bytes();files.append(dict(file=path.relative_to(out).as_posix(),bytes=len(blob),sha256=hashlib.sha256(blob).hexdigest()))
(out/'manifest.json').write_text(json.dumps(dict(status='proof_passed',source_fingerprint=source,files=files),sort_keys=True,indent=2)+'\n')
print('proof_passed',len(files),out)
