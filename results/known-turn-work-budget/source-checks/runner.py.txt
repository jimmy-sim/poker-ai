"""Serial exact-source base/optional qualification; persistent raw logs."""
import hashlib
import json
import os
import subprocess
import sys
import time
from pathlib import Path

root = Path(__file__).resolve().parents[3]
out = Path(__file__).resolve().parent
head = '674d7836f953b4f52e36cd4940351072dbdd384a'
source = '2a64f9e0e5cc620d50d43e595ab3e4e1e4b3414a729b5d487048f7a33e410d7a'
roles = ('base311', 'base312', 'optional311', 'optional312')
envs = root / 'results/runs/guarded-skip-requalification-20261118/envs'
suites = json.loads((out / 'optional-suites.json').read_text())
assert len(suites) == 82 and len(set(suites)) == 82

def guard(env):
    code = '''import hashlib,json,pathlib,subprocess,sys,numpy,poker_ai
r=pathlib.Path.cwd();sys.path.insert(0,str(r/'scripts'))
from known_turn_work_budget.gate import clean_revision,source_fingerprint,source_files
from known_turn_work_budget.plan import contract
assert clean_revision(r)==%r and source_fingerprint(r)==%r
assert len(source_files(r))>=16
assert contract(r)['charged_work_labels']==[10000000,20000000,40000000]
assert hashlib.sha256((r/'uv.lock').read_bytes()).hexdigest()=='db5529861ae56225daaf8a18513013e52bee88ef667227855b719a7e1364250c'
assert pathlib.Path(poker_ai.__file__).resolve().is_relative_to(r/'src')
assert 'PYTHONPATH' not in __import__('os').environ
print(json.dumps(dict(head=%r,source_fingerprint=%r,python=sys.version.split()[0],numpy=numpy.__version__,dirty=False)))
''' % (head, source, head, source)
    if env.startswith('optional'):
        code += '''from importlib.metadata import distribution,version
provider=distribution('open-spiel')
licenses=[f for f in provider.files or () if str(f).endswith('licenses/LICENSE')]
assert len(licenses)==1
raw=pathlib.Path(str(provider.locate_file(licenses[0]))).read_bytes()
assert version('numba')=='0.68.1' and version('open-spiel')=='2.0.2'
assert len(raw)==11358 and hashlib.sha256(raw).hexdigest()=='cfc7749b96f63bd31c3c42b5c471bf756814053e847c10f3eb003417bc523d30'
'''
    return code

for role in roles:
    minor = role[-3:]
    env = {k:v for k,v in os.environ.items() if k in {'PATH','HOME','TMPDIR','USER','LOGNAME','SHELL','LANG','LC_ALL','LC_CTYPE','UV_CACHE_DIR','UV_PYTHON_INSTALL_DIR','UV_OFFLINE','PRE_COMMIT_HOME'}}
    env.update(UV_PROJECT_ENVIRONMENT=str(envs/role),UV_PYTHON={'311':'3.11.17','312':'3.12.15'}[minor])
    prefix = ['uv','run','--locked'] + (['--all-extras'] if role.startswith('optional') else [])
    commands = [prefix+['python','-c',guard(role)], ['make','check'] if role.startswith('base') else prefix+['pytest','-q',*suites], prefix+['python','-c',guard(role)]]
    log = out/f'{role}.log'
    receipt = out/f'{role}.json'
    if log.exists() or receipt.exists():
        raise FileExistsError(f'retain prior role: {role}')
    start = time.perf_counter()
    failure = None
    try:
        with log.open('x') as stream:
            for i,cmd in enumerate(commands):
                stream.write(json.dumps(dict(command_index=i,command=cmd[:5],suite_count=len(suites) if role.startswith('optional') else None))+'\n')
                stream.flush()
                subprocess.run(cmd,cwd=root,env=env,stdout=stream,stderr=subprocess.STDOUT,check=True)
    except BaseException as error:
        failure = dict(type=type(error).__name__,reason=str(error))
    finally:
        receipt.write_text(json.dumps(dict(role=role,head=head,source_fingerprint=source,status='passed' if failure is None else 'failed',failure=failure,elapsed_seconds=time.perf_counter()-start,log_sha256=hashlib.sha256(log.read_bytes()).hexdigest()),indent=2)+'\n')
    print(role, 'passed' if failure is None else 'failed', flush=True)
    if failure is not None:
        sys.exit(1)
