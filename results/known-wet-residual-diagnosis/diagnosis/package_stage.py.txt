"""Seal the one preregistered saved-policy residual diagnosis and raw traces."""

import hashlib
import json
import shutil
import subprocess
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
RAW = Path(__file__).resolve().parent
OUT = ROOT / "results/known-wet-residual-diagnosis/diagnosis"
SOURCE = "4901950e9b49d2f8e383152e67dba06f82b70a486db19905b9e810fdd67ff2d3"


def main() -> None:
    if OUT.exists() or OUT.is_symlink():
        raise FileExistsError("retain earlier diagnosis")
    head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip()
    if subprocess.check_output(["git", "status", "--porcelain=v1"], cwd=ROOT):
        raise RuntimeError("diagnosis packaging requires clean head")
    state = json.loads((RAW / "job-state.json").read_bytes())
    report = json.loads((RAW / "raw/report.json").read_bytes())
    selection = json.loads((RAW / "raw/selection.json").read_bytes())
    raw_files = [path for path in (RAW / "raw").iterdir() if path.is_file()]
    if (
        state["status"] != "passed"
        or state["exit_code"] != 0
        or state["cap_seconds"] != 900
        or not 0 < state["observed_wall_seconds"] < 900
        or state["source_head"] != head
        or state["source_fingerprint"] != SOURCE
        or report["status"] != "completed"
        or report["source_head"] != head
        or report["source_fingerprint"] != SOURCE
        or selection["selected"] != report["selection"]
        or len(selection["selected"]["TURN"]) != 10
        or len(selection["selected"]["RIVER"]) != 10
        or len(report["cases"]) != 12
        or len(report["diagnostics"]) != 20
        or any(len(row["observations"]) != 12 for row in report["diagnostics"])
        or len(raw_files) != 28
        or (RAW / "raw/failure.json").exists()
        or (RAW / "launch.exit").read_text().strip() != "0"
    ):
        raise ArithmeticError("fixed diagnosis grid, source or cap failed")
    expected = {
        (arm, work, readout)
        for arm in ("signed_full_q8_turn_exact_river", "signed_full_exact_turn_river")
        for work in (100000, 1000000, 10000000)
        for readout in ("current", "linear_average")
    }
    actual = {(row["arm"], row["work_label"], row["readout"]) for row in report["cases"]}
    if actual != expected:
        raise ArithmeticError("fixed 12-case policy grid changed")
    if any(path.is_symlink() for path in (RAW / "raw").rglob("*")):
        raise ValueError("raw diagnosis symlink")
    shutil.copytree(RAW / "raw", OUT)
    for name in ("job-state.json", "launch.exit", "launch.log", "launcher.py", "package_stage.py"):
        source = RAW / name
        target = OUT / (name + ".txt" if name.endswith(".py") else name)
        shutil.copyfile(source, target)
    files = []
    for path in sorted(path for path in OUT.rglob("*") if path.is_file()):
        data = path.read_bytes()
        files.append(dict(file=path.relative_to(OUT).as_posix(), bytes=len(data), sha256=hashlib.sha256(data).hexdigest()))
    (OUT / "manifest.json").write_text(json.dumps(dict(status="diagnosis_completed", source_fingerprint=SOURCE, files=files), sort_keys=True, indent=2) + "\n")
    print("diagnosis_completed", len(files), OUT)


if __name__ == "__main__":
    main()
