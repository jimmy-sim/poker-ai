"""Seal the separate data-only conservative compute admission."""

import hashlib
import json
import shutil
import subprocess
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
RAW = Path(__file__).resolve().parent
OUT = ROOT / "results/known-wet-residual-diagnosis/admission"
SOURCE = "4901950e9b49d2f8e383152e67dba06f82b70a486db19905b9e810fdd67ff2d3"


def main() -> None:
    if OUT.exists() or OUT.is_symlink():
        raise FileExistsError("retain earlier admission")
    head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip()
    if subprocess.check_output(["git", "status", "--porcelain=v1"], cwd=ROOT):
        raise RuntimeError("admission packaging requires clean head")
    state = json.loads((RAW / "job-state.json").read_bytes())
    decision = json.loads((RAW / "raw/decision.json").read_bytes())
    if (
        state["status"] != "passed"
        or state["exit_code"] != 0
        or state["cap_seconds"] != 180
        or not 0 < state["observed_wall_seconds"] < 180
        or state["source_head"] != head
        or state["source_fingerprint"] != SOURCE
        or decision["status"] != "admitted"
        or decision["head"] != head
        or decision["source_fingerprint"] != SOURCE
        or decision["parent_report_wall_seconds"] != 68.30351770902053
        or decision["forecast_bound_seconds"] != 256.60703541804106
        or decision["hard_cap_seconds"] != 900
        or decision["forecast_bound_seconds"] >= decision["hard_cap_seconds"]
        or (RAW / "launch.exit").read_text().strip() != "0"
    ):
        raise ArithmeticError("data-only admission identity, formula or cap failed")
    if any(path.is_symlink() for path in (RAW / "raw").rglob("*")):
        raise ValueError("admission raw symlink")
    shutil.copytree(RAW / "raw", OUT)
    for name in ("job-state.json", "launch.exit", "launch.log", "launcher.py", "package_stage.py"):
        source = RAW / name
        target = OUT / (name + ".txt" if name.endswith(".py") else name)
        shutil.copyfile(source, target)
    files = []
    for path in sorted(path for path in OUT.rglob("*") if path.is_file()):
        data = path.read_bytes()
        files.append(dict(file=path.relative_to(OUT).as_posix(), bytes=len(data), sha256=hashlib.sha256(data).hexdigest()))
    (OUT / "manifest.json").write_text(json.dumps(dict(status="admitted", source_fingerprint=SOURCE, files=files), sort_keys=True, indent=2) + "\n")
    print("admitted", len(files), OUT)


if __name__ == "__main__":
    main()
