"""Seal the single fictional proof after source receipt publication."""

import hashlib
import json
import shutil
import subprocess
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
RAW = Path(__file__).resolve().parent
OUT = ROOT / "results/known-wet-residual-diagnosis/proof"
FAILED = ROOT / "results/runs/known-wet-residual-proof-20261124"
SOURCE = "4901950e9b49d2f8e383152e67dba06f82b70a486db19905b9e810fdd67ff2d3"


def main() -> None:
    if OUT.exists() or OUT.is_symlink():
        raise FileExistsError("retain earlier fictional proof")
    head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip()
    if subprocess.check_output(["git", "status", "--porcelain=v1"], cwd=ROOT):
        raise RuntimeError("proof packaging requires clean head")
    state = json.loads((RAW / "job-state.json").read_bytes())
    proof = json.loads((RAW / "raw/proof.json").read_bytes())
    first = json.loads((FAILED / "job-state.json").read_bytes())
    if (
        state["status"] != "passed"
        or state["exit_code"] != 0
        or state["cap_seconds"] != 180
        or not 0 < state["observed_wall_seconds"] < 180
        or state["source_head"] != head
        or state["source_fingerprint"] != SOURCE
        or proof["status"] != "proof_passed"
        or proof["head"] != head
        or proof["source_fingerprint"] != SOURCE
        or proof["fictional"] is not True
        or proof["policy_outcomes_read"] is not False
        or proof["turn_selection"] != [2, 0, 1, 3, 4, 5, 6, 7, 8, 9]
        or proof["river_selection"] != [14, 12, 13, 15, 16, 17, 18, 19, 20, 21]
        or (RAW / "launch.exit").read_text().strip() != "0"
        or first["status"] != "failed_retained"
        or first["exit_code"] != 2
        or (FAILED / "raw").exists()
        or "Required uv version `==0.13.5`" not in (FAILED / "launch.log").read_text()
    ):
        raise ArithmeticError("fictional proof identity or cap failed")
    if any(path.is_symlink() for path in (RAW / "raw").rglob("*")):
        raise ValueError("proof raw symlink")
    shutil.copytree(RAW / "raw", OUT)
    for name in ("job-state.json", "launch.exit", "launch.log", "launcher.py", "package_stage.py"):
        source = RAW / name
        target = OUT / (name + ".txt" if name.endswith(".py") else name)
        shutil.copyfile(source, target)
    for name in ("job-state.json", "launch.exit", "launch.log", "launcher.log", "launcher.py"):
        source = FAILED / name
        target = OUT / ("first-preflight-" + name + (".txt" if name.endswith(".py") else ""))
        shutil.copyfile(source, target)
    files = []
    for path in sorted(path for path in OUT.rglob("*") if path.is_file()):
        data = path.read_bytes()
        files.append(dict(file=path.relative_to(OUT).as_posix(), bytes=len(data), sha256=hashlib.sha256(data).hexdigest()))
    (OUT / "manifest.json").write_text(json.dumps(dict(status="proof_passed", source_fingerprint=SOURCE, files=files), sort_keys=True, indent=2) + "\n")
    print("proof_passed", len(files), OUT)


if __name__ == "__main__":
    main()
