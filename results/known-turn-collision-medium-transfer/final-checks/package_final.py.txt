"""Seal PR116 exact negative report-head qualification with all retained logs."""
import gzip
import hashlib
import json
import shutil
import subprocess
from pathlib import Path

root = Path(__file__).resolve().parents[3]
raw = Path(__file__).resolve().parent
out = root / 'results/known-turn-collision-medium-transfer/final-checks'
head = '8095d0c17b514d0b1f4e92ab87a45cb79109818e'
source = '4ab36412d18231aada57b9cc4a4e96c9491f3e37418f8794e7768d3d19073eb0'
roles = ('base311', 'base312', 'optional311', 'optional312')
assert not out.exists() and not out.is_symlink()
assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=root, text=True).strip() == head
assert not subprocess.check_output(['git', 'status', '--porcelain=v1'], cwd=root)
receipts = [json.loads((raw / f'{role}.json').read_bytes()) for role in roles]
assert all(x['status'] == 'passed' and x['head'] == head and x['source_fingerprint'] == source and x['failure'] is None for x in receipts)
for role, phrase in (('base311', '5854 passed, 73 skipped, 150 deselected'), ('base312', '5854 passed, 73 skipped, 150 deselected'), ('optional311', '2601 passed'), ('optional312', '2599 passed, 2 skipped')):
    assert phrase in (raw / f'{role}.log').read_text(), role
    assert hashlib.sha256((raw / f'{role}.log').read_bytes()).hexdigest() == receipts[roles.index(role)]['log_sha256']
assert (raw / 'exact-hooks.exit').read_text().strip() == '0'
hosted = json.loads((raw / 'hosted.json').read_bytes())
assert hosted['headSha'] == head and hosted['conclusion'] == 'success'
assert {j['name']: j['conclusion'] for j in hosted['jobs']} == {'Python 3.11': 'success', 'Python 3.12': 'success'}
report_bytes = (root / 'results/known-turn-collision-medium-transfer/report/report.json').read_bytes()
report = json.loads(report_bytes)
assert report['status'] == 'completed' and report['primary']['classification'] == 'negative'
assert len(report['cases']) == 48 and len(report['checkpoint_contrasts']) == 24 and len(report['diagnostics']) == 32
assert report['source_fingerprint'] == source and report['native_maximum_error_chips'] <= 1e-9
assert hashlib.sha256(report_bytes).hexdigest() == '971ee4049f1696e9b1e979aafc0885e4707928fb4312c2677fd97c72d3631589'
assert hashlib.sha256((root / 'docs/known-turn-collision-medium-transfer-result.md').read_bytes()).hexdigest() == '41f2fccea63c120cc77233a83883f4a3243e53e9cd67ed7b4a1acb30a0eace5b'
assert (raw / 'report-remote-verification.json').is_file()
out.mkdir(parents=True)

def copy(path, label):
    target = out / label
    if path.suffix == '.log':
        with target.with_name(target.name + '.gz').open('xb') as f:
            with gzip.GzipFile(filename='', mode='wb', fileobj=f, mtime=0) as z:
                z.write(path.read_bytes())
    elif path.suffix == '.py':
        shutil.copyfile(path, target.with_name(target.name + '.txt'))
    else:
        shutil.copyfile(path, target)

for name in ('runner.py', 'package_final.py', 'optional-suites.json', 'exact-hooks.exit', 'exact-hooks.log', 'hosted.json', 'hosted.log', 'report-remote-verification.json', 'pr-before.json', 'pr-body-report.md'):
    copy(raw / name, name)
for role in roles:
    for suffix in ('json', 'log'):
        copy(raw / f'{role}.{suffix}', f'{role}.{suffix}')
for folder, label in (('known-turn-collision-medium-transfer-final-20261124', 'launcher-sandbox-failure'), ('known-turn-collision-medium-transfer-final-20261124-permitted', 'launcher-short-head-failure')):
    source_dir = root / 'results/runs' / folder
    for name in ('base311.json', 'base311.log'):
        copy(source_dir / name, f'{label}-{name}')
result = dict(status='final_qualified', tested_head=head, source_fingerprint=source, report_classification='negative', local_roles=receipts, hosted_run=hosted['databaseId'], hosted_jobs=[dict(name=j['name'], conclusion=j['conclusion']) for j in hosted['jobs']], hooks='passed', operational_failures_retained=2)
(out / 'result.json').write_text(json.dumps(result, sort_keys=True, indent=2) + '\n')
files = []
for path in sorted(out.iterdir()):
    blob = path.read_bytes()
    files.append(dict(file=path.name, bytes=len(blob), sha256=hashlib.sha256(blob).hexdigest()))
(out / 'manifest.json').write_text(json.dumps(dict(status='final_qualified', source_fingerprint=source, files=files), sort_keys=True, indent=2) + '\n')
print('final_qualified', len(files), out)
