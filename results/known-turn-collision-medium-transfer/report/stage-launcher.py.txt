"""Run one registered PR116 stage with persistent job state and hard whole cap."""
import argparse
import json
import os
import signal
import subprocess
import time
from datetime import datetime, timezone
from pathlib import Path

parser = argparse.ArgumentParser()
parser.add_argument('stage', choices=('screen', 'proof', 'profile', 'admit', 'study', 'audit', 'report'))
parser.add_argument('--head', required=True)
parser.add_argument('--workdir', type=Path, required=True)
args = parser.parse_args()
root = Path(__file__).resolve().parents[2]
folder = args.workdir.resolve()
raw = root / 'results/known-turn-collision-medium-transfer/screen' if args.stage == 'screen' else folder / 'raw'
cap = {'screen':900,'proof':180,'profile':900,'admit':180,'study':7200,'audit':900,'report':3600}[args.stage]
source = '4ab36412d18231aada57b9cc4a4e96c9491f3e37418f8794e7768d3d19073eb0'
assert subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip() == args.head
assert not subprocess.check_output(['git','status','--porcelain=v1'],cwd=root)
assert not any(path.exists() or path.is_symlink() for path in (folder,raw))
if args.stage == 'screen':
    cmd = ['uv','run','--locked','python','scripts/prepare_known_turn_collision_medium_transfer.py','screen','--output',str(raw)]
elif args.stage in ('proof','profile','admit','study'):
    cmd = ['uv','run','--locked','python','scripts/study_known_turn_collision_medium_transfer.py',args.stage,'--output',str(raw)]
elif args.stage == 'audit':
    cmd = ['uv','run','--locked','--all-extras','python','scripts/audit_known_turn_collision_medium_transfer.py','--output',str(raw)]
else:
    cmd = ['uv','run','--locked','--all-extras','python','scripts/report_known_turn_collision_medium_transfer.py','--output',str(raw)]
folder.mkdir(parents=True, exist_ok=False)
log = folder / 'launch.log'
state = folder / 'job-state.json'
env = os.environ.copy(); env.pop('PYTHONPATH',None)
env['UV_PYTHON'] = '3.11.17'
env['UV_PROJECT_ENVIRONMENT'] = str(root/'results/runs/guarded-skip-requalification-20261118/envs'/('optional311' if args.stage in ('audit','report') else 'base311'))
started=time.perf_counter()
receipt=dict(status='starting',launcher_pid=os.getpid(),command=cmd,source_head=args.head,source_fingerprint=source,cap_seconds=cap,reserve_within_cap_seconds=120 if args.stage!='admit' else 0,environment={key:env[key] for key in ('UV_PYTHON','UV_PROJECT_ENVIRONMENT')},log=str(log),output=str(raw),started_utc=datetime.now(timezone.utc).isoformat())
state.write_text(json.dumps(receipt,sort_keys=True,indent=2)+'\n')
with log.open('xb') as stream:
    process=subprocess.Popen(cmd,cwd=root,env=env,stdout=stream,stderr=subprocess.STDOUT,start_new_session=True)
    receipt.update(status='running',child_pid=process.pid)
    state.write_text(json.dumps(receipt,sort_keys=True,indent=2)+'\n')
    try: code=process.wait(timeout=max(0.01,cap-(time.perf_counter()-started)))
    except subprocess.TimeoutExpired:
        os.killpg(process.pid,signal.SIGTERM)
        try: process.wait(timeout=5)
        except subprocess.TimeoutExpired: os.killpg(process.pid,signal.SIGKILL); process.wait()
        code=124
receipt.update(status='passed' if code==0 else 'failed_retained',exit_code=code,observed_wall_seconds=time.perf_counter()-started,finished_utc=datetime.now(timezone.utc).isoformat())
state.write_text(json.dumps(receipt,sort_keys=True,indent=2)+'\n')
(folder/'launch.exit').write_text(f'{code}\n')
print(json.dumps(receipt,sort_keys=True))
raise SystemExit(code)
