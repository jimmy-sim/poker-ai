"""Check raw publication bytes/accounting; do not calculate a cost comparison."""
import hashlib
import json
import shutil
from pathlib import Path

from compact_suit.cache import check_execution
from compact_suit.clock import measurement
from compact_suit.contract import contract, same
from compact_suit.source import check_source, source

w = Path(__file__).resolve().parent
r = w / "compact-suit"
base = r / "results/compact-suit-learning"
proof = w / "compact-suit-software-verified.json"
destination = base / "software/verified-publication.json"
assert not destination.exists()
shutil.copyfile(proof, destination)
rows = []
for rep in (0, 1, 2):
    folder = base / f"cost/{rep}"
    path = folder / "own.json"
    saved = json.loads(path.read_text())
    manifest = json.loads((base / f"raw-cost-own-{rep}-manifest.json").read_text())
    assert manifest["status"] == "complete_raw_before_analysis"
    assert len(manifest["raw_sha256"]) == 22
    for relative, expected in manifest["raw_sha256"].items():
        member = base / relative
        assert not member.is_symlink()
        assert hashlib.sha256(member.read_bytes()).hexdigest() == expected
        if relative.startswith(f"cost/{rep}/"):
            original = r / "results/runs/compact-suit" / relative
            assert member.read_bytes() == original.read_bytes()
    check_source(r, saved["source"])
    check_execution(r, saved)
    m = measurement(path, "cost-own")
    assert type(saved["repetition"]) is int and saved["repetition"] == rep
    assert saved["status"] == "completed_own_native_pending" and saved["failure"] is None
    assert same(saved["config"], contract(r))
    for key in ("reference_constructed", "reference_output_used_in_training", "poker_training_performed", "training_start_attempted", "new_agent_default", "fresh_global_admission", "long_run_admitted"):
        assert saved[key] is False
    assert saved["actual_poker_training_work"] == 0
    assert saved["own_replay"]["policy_and_ledger_bits_exact"] is True
    assert saved["own_replay"]["maximum_error"] == 0
    assert saved["own_replay"]["unique_profiles"] == 432
    assert saved["own_replay"]["fixed_conditions"] == 1152
    for name, expected in saved["report"].items():
        assert hashlib.sha256((folder / "report" / name).read_bytes()).hexdigest() == expected
    admin = json.loads((base / f"execution-cost-own-{rep}/exit.json").read_text())
    assert type(admin["returncode"]) is int and admin["returncode"] == 0
    assert admin["complete_command_within_cap"] is True and admin["interrupted"] is None
    assert 0 < admin["seconds"] < 900
    rows.append(dict(repetition=rep, raw_files=22, complete_seconds=m["seconds"],
                     supervised_outer_seconds=admin["seconds"], own_sha256=hashlib.sha256(path.read_bytes()).hexdigest(),
                     own_replay=saved["own_replay"], normalization=saved["normalization"]))
current = source(r)
assert len(current["sha256"]) == 433
assert current["fingerprint"] == "7381900fbd120938875a51bbd32ddb0afa340527eba5db1e1c50824a859e3f78"
result = dict(status="all_three_raw_own_costs_verified_before_publication_and_native",
              scope="Administrative source, raw bytes, saved replay/accounting and clocks; no new mathematical replay or comparative analysis here",
              original_ignored_outputs_retained=True, source_sha256=current["fingerprint"], rows=rows)
(base / "own-cost-raw-verification.json").open("x").write(json.dumps(result, indent=2, sort_keys=True) + "\n")
shutil.copyfile(__file__, base / "checks/operator_verify_own_cost_raw.py.txt")
print(json.dumps(dict(status=result["status"], repetitions=3, raw_files=66, source_files=433)))
