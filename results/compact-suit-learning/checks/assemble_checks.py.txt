"""Archive observed immutable-source role checks, with exact untrimmed logs."""

import hashlib, json, re, shutil, runpy, subprocess
from pathlib import Path

w = Path(__file__).resolve().parent
r = w / "compact-suit"
f = w / "compact-suit-source-v2-checks"
head = "066f5227a41613dde8e728936dac007b10f1e88f"
fp = "7381900fbd120938875a51bbd32ddb0afa340527eba5db1e1c50824a859e3f78"
run = json.loads((f / "run.json").read_text())["id"]
out = r / "results/compact-suit-learning/checks"
rows = []


def bind(p):
    return dict(
        file=str(p.relative_to(r)), sha256=hashlib.sha256(p.read_bytes()).hexdigest()
    )


for role in ("base", "optional"):
    for minor in ("311", "312"):
        name = role + minor
        x = json.loads((f / f"{name}-receipt.json").read_text())
        log = (f / f"{name}.log").read_text()
        assert (
            x["status"] == "passed"
            and x["checked_commit"] == head
            and x["source_fingerprint"] == fp
        )
        line = next(t for t in log.splitlines() if re.match(r"^\d+ passed", t))
        assert line.startswith(
            "4661 passed, 65 skipped, 150 deselected"
            if role == "base"
            else "1564 passed"
        )
        rows.append(
            dict(
                role=role,
                minor=minor,
                status="passed",
                source_fingerprint=fp,
                checked_commit=head,
                python="3.11.17" if minor == "311" else "3.12.15",
                numpy="2.4.6" if minor == "311" else "2.5.4",
                result=line,
            )
        )
m = json.loads((f / "run.json").read_text())
jobs = json.loads((f / "jobs.json").read_text())["jobs"]
assert (
    m["head_sha"] == head
    and m["status"] == "completed"
    and m["conclusion"] == "success"
    and m["id"] == run
)
for minor in ("311", "312"):
    job = next(
        x
        for x in jobs
        if x["name"] == "Python " + ("3.11" if minor == "311" else "3.12")
    )
    assert (
        job["status"] == "completed"
        and job["conclusion"] == "success"
        and job["head_sha"] == head
    )
    log = (f / f"hosted{minor}.log").read_text(encoding="utf-8-sig")
    assert head in log and "4661 passed, 65 skipped, 150 deselected" in log
    version = "3.11.17" if minor == "311" else "3.12.15"
    numpy = "2.4.6" if minor == "311" else "2.5.4"
    parse = runpy.run_path(
        str(r / "results/ci-runtime-correction/audit_checks.py.txt"),
        run_name="runtime_label_parser",
    )["parse"]
    actual = parse(log)
    assert actual["actual_python"] == [version] and actual["all_uv_versions"] == [
        version
    ]
    assert "ACTUAL_PYTHON=" + version in log and "ACTUAL_NUMPY=" + numpy in log
    guard = next(
        t for t in job["steps"] if t["name"] == "Verify actual installed runtime"
    )
    assert guard["conclusion"] == "success"
    checkout = json.loads((f / "checkout-source.json").read_text())
    tree = subprocess.check_output(
        ["git", "-C", str(r), "rev-parse", head + "^{tree}"], text=True
    ).strip()
    assert checkout["tree"]["sha"] == tree and checkout["sha"] in log
    assert head in [x["sha"] for x in checkout["parents"]]

    rows.append(
        dict(
            role="hosted",
            minor=minor,
            status="passed",
            result=next(line for line in log.splitlines() if "4661 passed, 65 skipped, 150 deselected" in line),
            source_fingerprint=fp,
            checked_commit=head,
            python="3.11.17" if minor == "311" else "3.12.15",
            numpy="2.4.6" if minor == "311" else "2.5.4",
            run_id=run,
            job_id=job["id"],
            head_sha=head,
            conclusion="success",
            runtime_verified_from_actual_guard=True,
            actual_uv_versions=actual["all_uv_versions"],
            actual_checkout_sha=checkout["sha"],
            actual_checkout_tree=tree,
            tested_tree_equals_pr_head=True,
        )
    )
assert not out.exists()
out.mkdir(parents=True)
for p in f.rglob("*"):
    if p.is_file():
        target = out / p.relative_to(f)
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(p, target)
shutil.copyfile(w / "run-compact-suit-source-v2-checks.py", out / "run_checks.py.txt")
shutil.copyfile(__file__, out / "assemble_checks.py.txt")
shutil.copyfile(w / "run-compact-suit-collection.py", out / "operator_launch.py.txt")
shutil.copyfile(w / "retain-compact-suit-collection.py", out / "operator_retention.py.txt")
shutil.copyfile(w / "publish-compact-suit-software.py", out / "operator_software_publication.py.txt")
shutil.copyfile(w / "check-compact-suit-operators.py", out / "check_operators.py.txt")
shutil.copyfile(w / "check-compact-suit-publication.py", out / "check_publication.py.txt")
shutil.copyfile(w / "compact-suit-publication-fixtures/checks.json", out / "publication-control-fixtures.json")
shutil.copytree(w / "compact-suit-dev-history", out / "development-history")
shutil.copyfile(w / "compact-suit-control-flow-review.json", out / "control-flow-self-review.json")
fixtures = out / "operator-fixtures"
fixtures.mkdir()
for name in ("checks.json", "success.stdout", "success.stderr", "nonzero.stdout", "nonzero.stderr", "timeout.stdout", "timeout.stderr", "interrupt.stdout", "interrupt.stderr"):
    shutil.copyfile(w / "compact-suit-operator-fixtures" / name, fixtures / name)
for x in rows:
    x["log"] = bind(out / f"{x['role']}{x['minor']}.log")
p = out.parent / "checks.json"
with p.open("x") as stream:
    json.dump(
        dict(
            source_fingerprint=fp,
            checks=rows,
            scope="Same-head433-source opt-in production-validation qualification; all418 predecessor source bytes preserved. Both full local bases, relevant all-extras suites and both hosted jobs bind actual runtime/all uv selections/complete tested trees. Suites may exercise toy/fixture own and reference solvers; they do not invoke this family's formal proof, complete real cost target or registered trajectory study. Superseded local cohort stays nonqualifying. Publish these six receipts and exact operators before the new complete shared180s own/native proof, all3 own costs before any native cost, full costs/forecast/checked admission. No default/fresh/global or playing-strength claim.",
        ),
        stream,
        indent=2,
    )
    stream.write("\n")
print(
    json.dumps(
        [
            dict(
                role=x["role"],
                minor=x["minor"],
                result=x.get("result"),
                job=x.get("job_id"),
            )
            for x in rows
        ]
    )
)
