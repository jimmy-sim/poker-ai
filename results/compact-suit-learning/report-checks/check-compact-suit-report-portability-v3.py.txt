"""Actual saved-data exports with all numerical experiment entry points blocked."""
import hashlib
import json
import sys
from pathlib import Path

import numpy
import pytest

w = Path(__file__).resolve().parent
r = w / "compact-suit"
sys.path.insert(0, str(r / "scripts"))
sys.path.insert(0, str(r / "tests"))
import report_suit_pipeline as report
from test_suit_pipeline_report import block_execution, blocked
from poker_ai.abstraction import potential
from poker_ai.postflop.matrix import MatrixTree
from poker_ai.postflop.tree import SpotTree
from poker_ai.postflop.controlled_scoring import ScoringContext
from poker_ai.solver_validation import opponent_learning
from compact_suit import cli
import suit_lookup_support as lookup

pair = (sys.version.split()[0], numpy.__version__)
assert pair in (("3.11.17", "2.4.6"), ("3.12.15", "2.5.4"))
minor = "311" if pair[0].startswith("3.11") else "312"
output = r / f"results/runs/compact-suit-report-v3-{minor}"
with pytest.MonkeyPatch.context() as monkeypatch:
    block_execution(monkeypatch)
    monkeypatch.setattr(MatrixTree, "build", classmethod(blocked))
    monkeypatch.setattr(SpotTree, "build", classmethod(blocked))
    monkeypatch.setattr(ScoringContext, "score", blocked)
    monkeypatch.setattr(potential, "canonical_cards", blocked)
    monkeypatch.setattr(lookup, "translated_cards", blocked)
    monkeypatch.setattr(opponent_learning, "audit", blocked)
    monkeypatch.setattr(cli, "_read", blocked)
    summary = report.render(r, output)
    report.show(summary)
exports = ("summary.json", "costs.csv", "forecast.csv")
if minor == "312":
    baseline = r / "results/runs/compact-suit-report-v3-311"
    assert all((output / name).read_bytes() == (baseline / name).read_bytes() for name in exports)
result = dict(status="saved_report_complete_with_execution_blocked", actual_python=pair[0], actual_numpy=pair[1],
              scope="Saved numeric portability; not real trajectory replay eligibility", numeric_exports=3,
              baseline_byte_equal=True if minor == "312" else None,
              blocked="Game construction/geometry, replay, training, independent workers, original/all/reduced normalization and installation; actual MatrixTree/SpotTree builders, ScoringContext.score, defining canonical/lookup functions, native audit and internal replay",
              raw_sha256={name: hashlib.sha256((output / name).read_bytes()).hexdigest() for name in exports})
(w / f"compact-suit-report-portability-v3-{minor}.json").open("x").write(json.dumps(result, indent=2, sort_keys=True) + "\n")
print(json.dumps(result))
