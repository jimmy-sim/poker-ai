"""Run one serial, exact-head optional qualification role with retained logs."""
import json
import os
import subprocess
import sys
import time
from pathlib import Path

root = Path('/Users/jimmysim/Documents/Codex/2026-11-02/files-pasted-by-the-user-role/work/receipt-consumer-mutation-checkout')
minor = sys.argv[1]
assert minor in ('311', '312')
version = '3.11.17' if minor == '311' else '3.12.15'
numpy = '2.4.6' if minor == '311' else '2.5.4'
head = 'dd53e9904d8d31cb0407a9662f0880f851907b7a'
fingerprint = 'b475bfefb92963f25458d97260264743ff4d751b4513c307529742eaf10fc97d'
paths = json.loads(Path('/private/tmp/receipt-consumer-mutation-optional-suites.json').read_text())
assert len(paths) == 68 and paths[-1] == 'tests/test_receipt_consumer_mutation.py'
allowed = {'PATH','HOME','TMPDIR','USER','LOGNAME','SHELL','LANG','LC_ALL','LC_CTYPE','UV_CACHE_DIR','UV_PYTHON_INSTALL_DIR','UV_OFFLINE','PRE_COMMIT_HOME','UV_PROJECT_ENVIRONMENT','UV_PYTHON'}
env = {k: v for k, v in os.environ.items() if k in allowed}
assert 'PYTHONPATH' not in env
cmd = ['uv', 'run', '--locked', '--all-extras']
guard = f'''import pathlib,subprocess,sys,numpy,poker_ai
r=pathlib.Path.cwd();sys.path.insert(0,str(r/'scripts'))
import profile_receipt_cpu as t
assert subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()=={head!r}
assert pathlib.Path(poker_ai.__file__).resolve().is_relative_to(r/'src')
assert sys.version.split()[0]=={version!r} and numpy.__version__=={numpy!r}
_,s=t.binding(r,clean=True)
assert len(s['sha256'])==458 and s['fingerprint']=={fingerprint!r} and s['dirty'] is False
assert 'PYTHONPATH' not in __import__('os').environ
print('ACTUAL_PYTHON='+sys.version.split()[0]);print('ACTUAL_NUMPY='+numpy.__version__)
print('CHECKED_COMMIT='+{head!r});print('SOURCE_FINGERPRINT='+s['fingerprint'])
'''
output = Path(f'/private/tmp/receipt-consumer-mutation-optional{minor}.log')
receipt = output.with_suffix('.json')
started = time.perf_counter()
status = 'failed'
try:
    with output.open('x') as log:
        for command in (cmd+['python','-c',guard], cmd+['pytest','-q',*paths], cmd+['python','-c',guard]):
            log.write(json.dumps({'command': command if len(command) < 10 else command[:6]+[f'{len(paths)} fixed suites']})+'\n')
            log.flush()
            subprocess.run(command,cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT,check=True)
    status = 'passed'
finally:
    receipt.write_text(json.dumps({'role':'optional'+minor,'status':status,'checked_commit':head,'python':version,'numpy':numpy,'source_fingerprint':fingerprint,'suite_count':len(paths),'elapsed_seconds':time.perf_counter()-started,'no_launcher_PYTHONPATH':True},indent=2)+'\n')
print(status, output)
