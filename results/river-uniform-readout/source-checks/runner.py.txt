"""Run exact-source locked local river qualification with retained logs."""
import hashlib
import json
import os
import subprocess
import sys
import time
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
OUT = Path(__file__).resolve().parent
ENVROOT = ROOT / "results/runs/guarded-skip-requalification-20261118/envs"
HEAD = "3e837b7e56c238d522de33bff0cacd8abcd16cf9"
FINGERPRINT = "a982d8c3537f9d50aa721001a0436766f34e862732b8e1b514fdc79355a963f6"
LOCK = "db5529861ae56225daaf8a18513013e52bee88ef667227855b719a7e1364250c"
minor, role = sys.argv[1:]
assert minor in ("311", "312") and role in ("base", "optional")
version, numpy = {"311": ("3.11.17", "2.4.6"), "312": ("3.12.15", "2.5.4")}[minor]
name = role + minor
env = {k: v for k, v in os.environ.items() if k in {
    "PATH", "HOME", "TMPDIR", "USER", "LOGNAME", "SHELL", "LANG", "LC_ALL",
    "LC_CTYPE", "UV_CACHE_DIR", "UV_PYTHON_INSTALL_DIR", "UV_OFFLINE", "PRE_COMMIT_HOME",
}}
env.update(UV_PROJECT_ENVIRONMENT=str(ENVROOT / name), UV_PYTHON=version)
assert "PYTHONPATH" not in env
suites = json.loads((OUT / "optional-suites.json").read_bytes())
assert len(suites) == 81 and len(set(suites)) == 81
assert all((ROOT / path).is_file() for path in suites)
guard = f'''import hashlib,json,pathlib,subprocess,sys,numpy,poker_ai
r=pathlib.Path.cwd();sys.path.insert(0,str(r/"scripts"))
from river_uniform_readout.gate import source_fingerprint,source_files,clean_revision
from river_generalization.gate import source_fingerprint as historical_fingerprint
assert historical_fingerprint(r)=="152b357b6942d065e402c84abc14ca3631b29d54551ead71bb668729ccbc81d3"
head=clean_revision(r)
assert head=={HEAD!r} and source_fingerprint(r)=={FINGERPRINT!r}
assert len(source_files(r))==397
from medium_dry_clipped.gate import fixed_contract, fingerprint, clean_head
assert clean_head(r)=={HEAD!r}
assert fingerprint(r)=="7c2312727903a44a347be5ea1bf5c6d1d2c1ed17c447bac7e3605fa8773320ab"
assert fixed_contract(r)["node_work_labels"]==[100000,1000000,10000000]
from medium_dry_b1.gate import fixed_contract as b1_contract, fingerprint as b1_fingerprint, clean_head as b1_clean
assert b1_clean(r)=={HEAD!r}
assert b1_fingerprint(r)=="64e92a742f865b6f19a87ad98e4b2ac3e94b2bf5ce8b26128fb2459a676669bd"
assert b1_contract(r)["node_work_labels"]==[100000,1000000,10000000]
from medium_dry_b1.gate import require_stage as b1_stage, OUTPUT as b1_output
assert b1_stage(r,b1_output/"report","report_negative")["status"]=="report_negative"
assert pathlib.Path(poker_ai.__file__).resolve().is_relative_to(r/'src')
assert hashlib.sha256((r/'uv.lock').read_bytes()).hexdigest()=={LOCK!r}
assert sys.version.split()[0]=={version!r} and numpy.__version__=={numpy!r}
assert 'PYTHONPATH' not in __import__('os').environ
print(json.dumps(dict(head=head,python=sys.version.split()[0],numpy=numpy.__version__,source_fingerprint=source_fingerprint(r),source_files=len(source_files(r)),lock_sha256={LOCK!r},dirty=False)))
'''
if role == "optional":
    guard += '''from importlib.metadata import distribution,version
provider=distribution("open-spiel")
licenses=[f for f in provider.files or () if str(f).endswith("licenses/LICENSE")]
assert len(licenses)==1
raw=pathlib.Path(str(provider.locate_file(licenses[0]))).read_bytes()
assert version("numba")=="0.68.1" and version("open-spiel")=="2.0.2"
assert len(raw)==11358 and hashlib.sha256(raw).hexdigest()=="cfc7749b96f63bd31c3c42b5c471bf756814053e847c10f3eb003417bc523d30"
print(json.dumps(dict(numba=version("numba"),open_spiel=version("open-spiel"),license_bytes=len(raw),license_sha256=hashlib.sha256(raw).hexdigest())))
'''
prefix = ["uv", "run", "--locked"]
if role == "optional":
    prefix.append("--all-extras")
commands = [prefix + ["python", "-c", guard],
            ["make", "check"] if role == "base" else prefix + ["pytest", "-q", *suites],
            prefix + ["python", "-c", guard]]
log = OUT / f"{name}.log"
receipt = OUT / f"{name}.json"
started = time.perf_counter()
status, failure = "failed", None
try:
    with log.open("x") as stream:
        for index, command in enumerate(commands):
            stream.write(json.dumps(dict(index=index, command=command if len(command) < 10 else command[:5], suite_count=len(suites) if role == "optional" else None)) + "\n")
            stream.flush()
            subprocess.run(command, cwd=ROOT, env=env, stdout=stream, stderr=subprocess.STDOUT, check=True)
    status = "passed"
except BaseException as error:
    failure = dict(type=type(error).__name__, reason=str(error))
    raise
finally:
    receipt.write_text(json.dumps(dict(role=name, status=status, head=HEAD, python=version,
        numpy=numpy, source_fingerprint=FINGERPRINT, lock_sha256=LOCK,
        suite_count=len(suites) if role == "optional" else None,
        elapsed_seconds=time.perf_counter() - started, failure=failure), indent=2) + "\n")
print(name, status, log)
