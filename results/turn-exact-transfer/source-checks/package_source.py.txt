"""Seal exact-source qualification with retained operator failures."""

import gzip
import hashlib
import json
import shutil
import subprocess
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
RAW = Path(__file__).resolve().parent
OUT = ROOT / "results/turn-exact-transfer/source-checks"
HEAD = "99dff043d08f51bd00a00acb39fa153822cd376e"
FINGERPRINT = "fb9006517142ee23076f2c0b6d581475dfdaed8127cb2c146a60731af901b09c"
ROLES = ("base311", "base312", "optional311", "optional312")


def copy(source: Path, destination: Path) -> None:
    if not source.is_file():
        raise FileNotFoundError(source)
    destination.parent.mkdir(parents=True, exist_ok=True)
    if source.suffix == ".log":
        with destination.with_name(destination.name + ".gz").open("xb") as stream:
            with gzip.GzipFile(filename="", mode="wb", fileobj=stream, mtime=0) as zipped:
                zipped.write(source.read_bytes())
    elif source.suffix == ".py":
        shutil.copyfile(source, destination.with_name(destination.name + ".txt"))
    else:
        shutil.copyfile(source, destination)


def main() -> None:
    if OUT.exists() or OUT.is_symlink():
        raise FileExistsError("retain prior source receipt")
    head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip()
    dirty = subprocess.check_output(["git", "status", "--porcelain=v1"], cwd=ROOT)
    if head != HEAD or dirty:
        raise ArithmeticError("exact tested source changed or dirty")
    state = json.loads((RAW / "driver-state.json").read_bytes())
    receipts = [json.loads((RAW / f"{role}.json").read_bytes()) for role in ROLES]
    hosted = json.loads((RAW / "hosted.json").read_bytes())
    verified = json.loads((RAW / "github-tree-verify.json").read_bytes())
    if state["status"] != "passed" or state["head"] != HEAD or any(
        row != {"role": role, "exit_code": 0}
        for row, role in zip(state["completed"], ROLES, strict=True)
    ):
        raise ArithmeticError("four serial exact-source roles incomplete")
    if any(
        row["status"] != "passed"
        or row["head"] != HEAD
        or row["source_fingerprint"] != FINGERPRINT
        for row in receipts
    ) or any(row["suite_count"] != 81 for row in receipts[2:]):
        raise ArithmeticError("local role source or optional suite mismatch")
    if (
        hosted["headSha"] != HEAD
        or hosted["conclusion"] != "success"
        or {row["name"] for row in hosted["jobs"]} != {"Python 3.11", "Python 3.12"}
        or any(row["conclusion"] != "success" for row in hosted["jobs"])
    ):
        raise ArithmeticError("both tested-source hosted jobs required")
    if (
        verified["status"] != "verified"
        or verified["head"] != HEAD
        or verified["source_fingerprint"] != FINGERPRINT
        or verified["count"] != 24
    ):
        raise ArithmeticError("GitHub closed source bytes not verified")
    if (RAW / "driver.exit").read_text().strip() != "0" or any(
        (RAW / f"{name}.exit").read_text().strip() != "0"
        for name in ("hooks", "focused")
    ) or "11 passed" not in (RAW / "focused.log").read_text():
        raise ArithmeticError("driver, hooks or focused checks incomplete")
    if (
        "5795 passed, 73 skipped, 150 deselected" not in (RAW / "base311.log").read_text()
        or "5795 passed, 73 skipped, 150 deselected" not in (RAW / "base312.log").read_text()
        or "2597 passed" not in (RAW / "optional311.log").read_text()
        or "2595 passed, 2 skipped" not in (RAW / "optional312.log").read_text()
    ):
        raise ArithmeticError("complete local test counts absent")

    OUT.mkdir(parents=True)
    names = [
        "driver-state.json", "driver.exit", "driver.log", "driver.py", "runner.py",
        "optional-suites.json", "package_source.py", "hooks.exit", "hooks.log",
        "focused.exit", "focused.log", "hosted.json", "hosted.log",
        "github-tree-verify.json",
    ]
    names.extend(f"{role}.{extension}" for role in ROLES for extension in ("json", "log"))
    for name in names:
        copy(RAW / name, OUT / name)
    failures: list[str] = []
    for label, folder in (
        ("first-launcher", ROOT / "results/runs/turn-exact-transfer-source-20261123"),
        ("first-guard", ROOT / "results/runs/turn-exact-transfer-source-20261123-envfix"),
    ):
        for name in ("driver.exit", "driver.log", "driver-state.json", "base311.json", "base311.log", "driver.py", "runner.py"):
            copy(folder / name, OUT / "failures" / label / name)
            archived = name + (".gz" if name.endswith(".log") else ".txt" if name.endswith(".py") else "")
            failures.append(f"failures/{label}/{archived}")
    iteration = ROOT / "results/runs/turn-exact-transfer-20261123"
    for source in sorted(iteration.glob("first-*")):
        if source.is_file():
            copy(source, OUT / "failures" / "iteration" / source.name)
            archived = source.name + (".gz" if source.suffix == ".log" else "")
            failures.append(f"failures/iteration/{archived}")
    copy(iteration / "full-source-mypy.log", OUT / "full-source-mypy.log")
    result = dict(
        status="source_qualified",
        tested_head=HEAD,
        source_fingerprint=FINGERPRINT,
        hosted_run=hosted["databaseId"],
        hosted_jobs=[{"name": row["name"], "conclusion": row["conclusion"]} for row in hosted["jobs"]],
        local_roles=receipts,
        focused_tests="11 passed on locked base311",
        full_strict_mypy="994 source files",
        hooks="passed",
        retained_failures=failures,
        selected_tree_or_outcome_measured=False,
    )
    (OUT / "result.json").write_text(json.dumps(result, sort_keys=True, indent=2) + "\n")
    files = []
    for path in sorted(OUT.rglob("*")):
        if path.is_file():
            raw = path.read_bytes()
            files.append(dict(file=path.relative_to(OUT).as_posix(), bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest()))
    (OUT / "manifest.json").write_text(
        json.dumps(dict(status="source_qualified", source_fingerprint=FINGERPRINT, files=files), sort_keys=True, indent=2) + "\n"
    )
    print(len(files), sum(row["bytes"] for row in files), OUT)


if __name__ == "__main__":
    main()
