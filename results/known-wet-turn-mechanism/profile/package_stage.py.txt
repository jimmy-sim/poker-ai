"""Archive all six fixed profile paths, repeats, forecast and launcher records."""

import hashlib
import json
import shutil
from pathlib import Path

SOURCE = "5e29c490f3b05cf7b98dc016628a9765ab76b7a920965590e0d086abcda16b12"
HEAD = "6e69187ee3103221438155e44accbd3094d154e0"


def package(folder: Path, output: Path) -> None:
    if output.exists() or output.is_symlink():
        raise FileExistsError("retain earlier fixed profile")
    state = json.loads((folder / "job-state.json").read_bytes())
    result = json.loads((folder / "raw/result.json").read_bytes())
    forecast = json.loads((folder / "raw/forecast.json").read_bytes())
    if (
        state["status"] != "passed"
        or state["exit_code"] != 0
        or state["cap_seconds"] != 900
        or not 0 < state["observed_wall_seconds"] < 900
        or state["source_head"] != HEAD
        or state["source_fingerprint"] != SOURCE
        or result["status"] != "completed"
        or result["stage"] != "profile"
        or result["head"] != HEAD
        or result["source_fingerprint"] != SOURCE
        or result["worker_count"] != 2
        or result["assignments"] != [[0, 2, 4], [1, 3, 5]]
        or len(forecast["per_path_maxima"]) != 2
    ):
        raise ArithmeticError("fixed profile identity, grid or cap failed")
    if (folder / "launch.exit").read_text().strip() != "0":
        raise ArithmeticError("profile launcher exit changed")
    if any(path.is_symlink() for path in (folder / "raw").rglob("*")):
        raise ValueError("raw profile symlink")
    shutil.copytree(folder / "raw", output)
    for name in ("job-state.json", "launch.exit", "launch.log", "launcher.log", "launcher.py", "package_stage.py"):
        shutil.copyfile(folder / name, output / (name + ".txt" if name.endswith(".py") else name))
    files = []
    for path in sorted(p for p in output.rglob("*") if p.is_file()):
        raw = path.read_bytes()
        files.append(dict(file=path.relative_to(output).as_posix(), bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest()))
    (output / "manifest.json").write_text(json.dumps(dict(status="profile_published", source_fingerprint=SOURCE, files=files),sort_keys=True,indent=2)+"\n")
    print("profile_published",len(files),output)


if __name__ == "__main__":
    root = Path(__file__).resolve().parents[3]
    package(Path(__file__).resolve().parent, root / "results/known-wet-turn-mechanism/profile")
