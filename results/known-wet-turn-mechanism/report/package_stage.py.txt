"""Seal the fixed negative known-wet decision report from published inputs."""

import hashlib
import json
import math
import shutil
from pathlib import Path

SOURCE = "5e29c490f3b05cf7b98dc016628a9765ab76b7a920965590e0d086abcda16b12"
HEAD = "4764728be302b42cfa6bbcfb3edea6cbbe60e814"


def package(folder: Path, output: Path) -> None:
    if output.exists() or output.is_symlink():
        raise FileExistsError("retain earlier fixed report")
    state = json.loads((folder / "job-state.json").read_bytes())
    report = json.loads((folder / "raw/report.json").read_bytes())
    primary = report["primary"]
    if (
        state["status"] != "passed"
        or state["exit_code"] != 0
        or state["source_head"] != HEAD
        or state["source_fingerprint"] != SOURCE
        or not 0 < state["observed_wall_seconds"] < 900
        or report["status"] != "completed"
        or report["head"] != HEAD
        or report["source_fingerprint"] != SOURCE
        or len(report["cases"]) != 12
        or len(report["checkpoint_contrasts"]) != 6
        or primary["classification"] != "negative"
        or primary["candidate_absolute_pass"] is not False
        or report["native_maximum_error_chips"] > 1e-9
        or not math.isclose(primary["candidate"] - primary["control"], primary["candidate_minus_control"], abs_tol=1e-12)
        or (folder / "launch.exit").read_text().strip() != "0"
    ):
        raise ArithmeticError("fixed negative report or publication identity changed")
    if not report["diagnostics"] or any(len(row["observations"]) != 12 for row in report["diagnostics"]):
        raise ArithmeticError("fixed both-readout decision grid incomplete")
    output.mkdir(parents=True)
    shutil.copyfile(folder / "raw/report.json", output / "report.json")
    for name in ("job-state.json", "launch.exit", "launch.log", "launcher.log", "launcher.py", "package_stage.py"):
        shutil.copyfile(folder / name, output / (name + ".txt" if name.endswith(".py") else name))
    files = []
    for path in sorted(p for p in output.rglob("*") if p.is_file()):
        raw = path.read_bytes()
        files.append(dict(file=path.relative_to(output).as_posix(), bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest()))
    (output / "manifest.json").write_text(json.dumps(dict(status="report_negative", source_fingerprint=SOURCE, files=files), sort_keys=True, indent=2) + "\n")
    print("report_negative", len(files), output)


if __name__ == "__main__":
    root = Path(__file__).resolve().parents[3]
    package(Path(__file__).resolve().parent, root / "results/known-wet-turn-mechanism/report")
