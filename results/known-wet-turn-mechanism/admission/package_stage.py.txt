"""Archive separate strict known-turn admission from published profile bytes."""

import hashlib
import json
import shutil
from pathlib import Path

SOURCE = "5e29c490f3b05cf7b98dc016628a9765ab76b7a920965590e0d086abcda16b12"
HEAD = "5048c0e5db8c9991c5552f9a802151d65fa2b742"
PROFILE_SHA256 = "7da55a27e0f94e9117217897146a1f29192b4ae87f20822d7c811c0b9b2306b8"


def package(folder: Path, output: Path) -> None:
    if output.exists() or output.is_symlink():
        raise FileExistsError("retain earlier admission")
    root = folder.parents[2]
    state = json.loads((folder / "job-state.json").read_bytes())
    decision = json.loads((folder / "raw/decision.json").read_bytes())
    receipt = json.loads((folder / "raw/receipt.json").read_bytes())
    if hashlib.sha256((root / "results/known-wet-turn-mechanism/profile/manifest.json").read_bytes()).hexdigest() != PROFILE_SHA256:
        raise ArithmeticError("published profile manifest changed")
    if (
        state["status"] != "passed"
        or state["exit_code"] != 0
        or state["source_head"] != HEAD
        or state["source_fingerprint"] != SOURCE
        or not 0 < state["observed_wall_seconds"] < 900
        or receipt["head"] != HEAD
        or receipt["source_fingerprint"] != SOURCE
        or receipt["decision_status"] != "admitted"
        or decision["status"] != "admitted"
        or decision["strict_bound_seconds"] >= 7200
        or decision["rule"] != "2*(2*T+600)<7200"
    ):
        raise ArithmeticError("separate admission identity or strict bound failed")
    if (folder / "launch.exit").read_text().strip() != "0":
        raise ArithmeticError("admission launcher exit changed")
    shutil.copytree(folder / "raw", output)
    for name in ("job-state.json", "launch.exit", "launch.log", "launcher.log", "launcher.py", "package_stage.py"):
        shutil.copyfile(folder / name, output / (name + ".txt" if name.endswith(".py") else name))
    files = []
    for path in sorted(p for p in output.rglob("*") if p.is_file()):
        raw = path.read_bytes()
        files.append(dict(file=path.relative_to(output).as_posix(), bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest()))
    (output / "manifest.json").write_text(json.dumps(dict(status="admitted", source_fingerprint=SOURCE, files=files),sort_keys=True,indent=2)+"\n")
    print("admitted",len(files),output)


if __name__ == "__main__":
    root = Path(__file__).resolve().parents[3]
    package(Path(__file__).resolve().parent, root / "results/known-wet-turn-mechanism/admission")
