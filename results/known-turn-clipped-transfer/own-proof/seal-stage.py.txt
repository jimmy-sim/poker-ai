"""Seal one complete PR121 stage without changing its raw scientific bytes."""
import argparse
import hashlib
import json
import shutil
import sys
from pathlib import Path
from typing import Any

parser=argparse.ArgumentParser()
parser.add_argument('stage',choices=('proof','profile','admit','study','audit-admit','audit','report'))
parser.add_argument('--workdir',type=Path,required=True)
args=parser.parse_args()
root=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(root/'scripts'))
from known_turn_clipped_transfer.forecast import forecast
from known_turn_clipped_transfer.gate import ADMISSION,AUDIT_ADMISSION,NATIVE,OWN,PROFILE,PROOF,REPORT,clean_head,source_fingerprint
from known_turn_clipped_transfer.native import audit_forecast
from known_turn_clipped_transfer.plan import EXPECTED_WORK,LABELS,paths

source='97db9310f7f0941111bce2c982fce455b94af3d1413f83703fde958bf9a21af8'
assert source_fingerprint(root)==source
head=clean_head(root)
folder=args.workdir.resolve();raw=folder/'raw'
out=root/{'proof':PROOF,'profile':PROFILE,'admit':ADMISSION,'study':OWN,'audit-admit':AUDIT_ADMISSION,'audit':NATIVE,'report':REPORT}[args.stage]
state:dict[str,Any]=json.loads((folder/'job-state.json').read_bytes())
assert state['status']=='passed' and state['exit_code']==0 and (folder/'launch.exit').read_text().strip()=='0'
assert state['source_fingerprint']==source and state['source_head']==head and 0<state['observed_wall_seconds']<state['cap_seconds']
assert state['cap_seconds']=={'proof':180,'profile':900,'admit':180,'study':7200,'audit-admit':180,'audit':900,'report':3600}[args.stage]
assert raw.is_dir() and not raw.is_symlink() and not out.exists() and not out.is_symlink()
if args.stage in ('admit','audit-admit'):
 result=json.loads((raw/'decision.json').read_bytes())
 assert result['status']=='admitted'
 if args.stage=='admit': assert result==forecast(json.loads((root/PROFILE/'result.json').read_bytes()))
 else: assert result==audit_forecast(root)
 status='admitted'
elif args.stage=='report':
 result=json.loads((raw/'report.json').read_bytes())
 assert result['status']=='completed' and result['source_fingerprint']==source and result['head']==head
 assert len(result['cases'])==48 and len(result['checkpoint_contrasts'])==24 and len(result['diagnostics'])==42
 assert result['primary']['classification'] in ('positive','negative','inconclusive')
 status='report_'+result['primary']['classification']
else:
 result=json.loads((raw/'result.json').read_bytes())
 assert result['source_fingerprint']==source and result['head']==head and result['status']=='completed'
 if args.stage=='proof':
  assert result['stage']=='proof' and result['captures']==6 and result['real_policy_opened'] is False
  assert max(result['signed_error'],result['clipped_error'],result['average_error'])<=1e-12
  assert result['negative_signed_entries']>0 and result['negative_clipped_entries']==0
  status='proof_passed'
 elif args.stage=='profile':
  assert result['stage']=='profile' and forecast(result)==json.loads((raw/'forecast.json').read_bytes())
  status='profile_published'
 elif args.stage=='study':
  assert result['stage']=='study' and result['assignments']==[list(range(0,4,2)),list(range(1,4,2))]
  for spec in paths():
   saved=json.loads((raw/'study'/f'path{spec.index:02d}'/'summary.json').read_bytes())
   assert saved['status']=='completed' and saved['path']==dict(index=spec.index,game=spec.game,arm=spec.arm,seed=0)
   assert saved['labels']==list(LABELS) and len(saved['captures'])==6 and saved['recall_violations']==0
   assert [next(c['metadata']['total_work'] for c in saved['captures'] if c['metadata']['work_label']==label) for label in LABELS]==list(EXPECTED_WORK)
  status='own_published'
 else:
  assert len(result['rows'])==24 and result['maximum_error_chips']<1e-9
  status='native_audited'
shutil.copytree(raw,out)
for name in ('job-state.json','launch.exit','launch.log'):
 shutil.copyfile(folder/name,out/name)
shutil.copyfile(Path(__file__),out/'seal-stage.py.txt')
shutil.copyfile(root/'results/runs/known-turn-clipped-transfer-stage-launcher.py',out/'stage-launcher.py.txt')
files=[]
for path in sorted(p for p in out.rglob('*') if p.is_file() and p.name!='manifest.json'):
 assert not path.is_symlink()
 blob=path.read_bytes()
 files.append(dict(file=path.relative_to(out).as_posix(),bytes=len(blob),sha256=hashlib.sha256(blob).hexdigest()))
assert not (out/'manifest.json').exists()
(out/'manifest.json').write_text(json.dumps(dict(status=status,source_fingerprint=source,files=files),sort_keys=True,indent=2)+'\n')
print(status,len(files),out,hashlib.sha256((out/'manifest.json').read_bytes()).hexdigest())
