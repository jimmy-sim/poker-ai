for role in roles:
    minor = role[-3:]
    env = {k:v for k,v in os.environ.items() if k in {'PATH','HOME','TMPDIR','USER','LOGNAME','SHELL','LANG','LC_ALL','LC_CTYPE','UV_CACHE_DIR','UV_PYTHON_INSTALL_DIR','UV_OFFLINE','PRE_COMMIT_HOME'}}
    env.update(UV_PROJECT_ENVIRONMENT=str(envs/role),UV_PYTHON={'311':'3.11.17','312':'3.12.15'}[minor])
    prefix = ['uv','run','--locked'] + (['--all-extras'] if role.startswith('optional') else [])
    commands = [prefix+['python','-c',guard(role)], ['make','check'] if role.startswith('base') else prefix+['pytest','-q',*suites], prefix+['python','-c',guard(role)]]
    log = out/f'{role}.log'
    receipt = out/f'{role}.json'
    if log.exists() or receipt.exists():
        raise FileExistsError(f'retain prior role: {role}')
    start = time.perf_counter()
    failure = None
    try:
        with log.open('x') as stream:
            for i,cmd in enumerate(commands):
                stream.write(json.dumps(dict(command_index=i,command=cmd[:5],suite_count=len(suites) if role.startswith('optional') else None))+'\n')
                stream.flush()
                subprocess.run(cmd,cwd=root,env=env,stdout=stream,stderr=subprocess.STDOUT,check=True)
    except BaseException as error:
        failure = dict(type=type(error).__name__,reason=str(error))
    finally:
        receipt.write_text(json.dumps(dict(role=role,head=head,source_fingerprint=source,status='passed' if failure is None else 'failed',failure=failure,elapsed_seconds=time.perf_counter()-start,log_sha256=hashlib.sha256(log.read_bytes()).hexdigest()),indent=2)+'\n')
    print(role, 'passed' if failure is None else 'failed', flush=True)
    if failure is not None:
        sys.exit(1)
