"""Package exact B8 source checks and the retained pre-commit environment failure."""
import gzip
import hashlib
import json
import shutil
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
RAW = Path(__file__).resolve().parent
OUT = ROOT / 'results/medium-dry-b8-updates/source-checks'
HEAD = '81ef11afffd2067e94b3c576e5d2cb3e76f733cc'
FINGERPRINT = '666ac9e4322ca84f93ad43acf972e8a758125f1aa1276da2847a6025049b40ad'


def copy(source: Path, target: Path) -> None:
    assert source.is_file() and not target.exists(), (source, target)
    target.parent.mkdir(parents=True, exist_ok=True)
    if source.suffix == '.log':
        target = target.with_suffix('.log.gz')
        with target.open('wb') as stream:
            with gzip.GzipFile(filename='', mode='wb', fileobj=stream, mtime=0) as zipped:
                zipped.write(source.read_bytes())
    elif source.suffix == '.py':
        shutil.copyfile(source, target.with_name(target.name + '.txt'))
    else:
        shutil.copyfile(source, target)


if __name__ == '__main__':
    assert not OUT.exists()
    state = json.loads((RAW / 'driver-state.json').read_bytes())
    roles = ('base311', 'base312', 'optional311', 'optional312')
    receipts = [json.loads((RAW / f'{role}.json').read_bytes()) for role in roles]
    hosted = json.loads((RAW / 'hosted.json').read_bytes())
    first = json.loads((RAW / 'first-commit-hook-failure.json').read_bytes())
    assert state['head'] == HEAD and state['status'] == 'passed'
    assert all(row['status'] == 'passed' and row['head'] == HEAD for row in receipts)
    assert all(row['suite_count'] == 81 for row in receipts if row['role'].startswith('optional'))
    assert (RAW / 'hooks.exit').read_text().strip() == '0'
    assert hosted['headSha'] == HEAD and hosted['conclusion'] == 'success'
    assert {job['name'] for job in hosted['jobs']} == {'Python 3.11', 'Python 3.12'}
    assert all(job['conclusion'] == 'success' for job in hosted['jobs'])
    assert first['status'] == 'failed_before_commit' and not first['commit_created']
    assert '8 passed' in (RAW/'focused-optional311.log').read_text()
    assert '8 passed' in (RAW/'focused-optional312.log').read_text()
    OUT.mkdir(parents=True)
    names = ('driver-state.json', 'driver.log', 'driver.py', 'runner.py',
             'optional-suites.json', 'package_source.py', 'package_stage.py', 'hooks.exit', 'hooks.log',
             'hosted.json', 'hosted.log', 'first-commit-hook-failure.json',
             'hooks-first.log', 'hooks-first-shell-failure.json',
             'focused-optional311.log', 'focused-optional312.log')
    for name in names:
        copy(RAW / name, OUT / name)
    for role in roles:
        for ext in ('json', 'log'):
            copy(RAW / f'{role}.{ext}', OUT / f'{role}.{ext}')
    result = dict(status='source_qualified', tested_head=HEAD,
                  source_fingerprint=FINGERPRINT,
                  historical_fingerprint='152b357b6942d065e402c84abc14ca3631b29d54551ead71bb668729ccbc81d3',
                  parent_fingerprint='7c2312727903a44a347be5ea1bf5c6d1d2c1ed17c447bac7e3605fa8773320ab',
                  local_roles=receipts, hooks='passed', hosted_run=hosted['databaseId'],
                  hosted_jobs=[{'name': j['name'], 'conclusion': j['conclusion']} for j in hosted['jobs']],
                  first_hook_environment_failure='first-commit-hook-failure.json',
                  scientific_measurements_started=False)
    (OUT / 'result.json').write_text(json.dumps(result, sort_keys=True, indent=2) + '\n')
    files=[]
    for path in sorted(item for item in OUT.rglob('*') if item.is_file()):
        raw = path.read_bytes()
        files.append(dict(file=path.relative_to(OUT).as_posix(), bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest()))
    (OUT / 'manifest.json').write_text(json.dumps(dict(status='source_qualified', source_fingerprint=FINGERPRINT, files=files), sort_keys=True, indent=2) + '\n')
