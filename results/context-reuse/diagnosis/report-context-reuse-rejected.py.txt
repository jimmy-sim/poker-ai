"""Post-outcome diagnosis of a publicly retained REJECTED comparison.

This requires the original failure and never changes its admission status.
The child/source/raw/journal/clock checks are derived from frozen C0 saved_pair;
only top-level status handling requires failure and verifies original rejection.
No poker construction, replay, solver or new measurement is permitted.
"""

import argparse
import ast
import copy
import csv
import hashlib
import importlib
import json
import math
import pkgutil
import sys
from contextlib import ExitStack
from pathlib import Path
from typing import Any
from unittest.mock import patch

SOURCE_SHA = "f4d47807e8c0e9fe898d54fe84644fca46691e6b02a0973aa9941bc63104d1b2"
WORK_PATHS = (
    ("normalization", "requests"),
    ("normalization", "reduced_evaluations"),
    ("validation_cache", "requests"),
    ("validation_cache", "hits"),
)


def read_rejected_pair(root: Path, output: Path) -> dict[str, Any]:
    """Authenticate complete saved evidence without a replay or reference call."""
    from checkpoint_cpu_support import check_clock
    from compact_suit.cache import check_execution
    from profile_compact_replay import authenticate_source

    from poker_ai.evaluation.opponent_learning_workers import gather_independent
    from poker_ai.evaluation.private_prefix_journal import validate as validate_journal
    from poker_ai.postflop.learning_artifacts import sha
    from poker_ai.solver_validation.opponent_learning import validate_rows

    c = contract(root)
    pair_data = json.loads((output / "pair.json").read_text())
    if (
        pair_data["admission_evidence"] is not False
        or pair_data["order"] != c["order"]
        or len(pair_data["measurements"]) != 2
        or pair_data["status"] != "failed_retained"
        or pair_data.get("failure_kind") != "ArithmeticError"
    ):
        raise ArithmeticError("complete fixed ordered pair required")
    children, ledgers = [], []
    for mode, measurement in zip(c["order"], pair_data["measurements"], strict=True):
        folder = output / mode
        child_data = json.loads((folder / "child.json").read_text())
        if (
            measurement != json.loads((folder / "measurement.json").read_text())
            or child_data["admission_evidence"] is not False
            or child_data["reference_constructed"] is not True
            or child_data["poker_training_performed"] is not False
            or child_data["reference_output_used_in_training"] is not False
            or measurement["returncode"] != 0
            or type(measurement["returncode"]) is not int
            or measurement["interrupted"] is not None
            or type(measurement["seconds"]) not in (int, float)
            or type(measurement["cap_seconds"]) not in (int, float)
            or not math.isfinite(measurement["seconds"])
            or not math.isfinite(measurement["cap_seconds"])
            or not 0 < measurement["cap_seconds"] <= 900
            or not 0 < measurement["seconds"] < min(900, measurement["cap_seconds"])
            or measurement["child_sha256"] != digest(folder / "child.json")
            or type(measurement["child_pid"]) is not int
            or measurement["child_pid"] != child_data["pid"]
            or child_data["mode"] != mode
            or child_data["source"]["fingerprint"]
            != sha(child_data["source"]["sha256"])
        ):
            raise ArithmeticError("saved complete producer measurement changed")
        for stamp in (child_data["body_before"], child_data["body_after"]):
            check_clock(stamp)
            if (
                stamp["pid"] != measurement["child_pid"]
                or stamp["thread"] != child_data["thread"]
            ):
                raise ArithmeticError("complete body producer changed")
        start, end = child_data["body_before"], child_data["body_after"]
        if (
            not measurement["started_perf_counter"]
            <= start["wall"]
            <= end["wall"]
            <= measurement["started_perf_counter"] + measurement["seconds"]
            or child_data["complete_body_parent_cpu_seconds"]
            != end["cpu"] - start["cpu"]
            or end["cpu"] < start["cpu"]
        ):
            raise ArithmeticError("complete body and supervisor clocks changed")
        expected_command = [
            child_data["executable"],
            str(Path(child_data["execution_root"]) / ENTRY),
            "--child",
            mode,
            "--output",
            child_data["execution_output"],
        ]
        if measurement["command"] != expected_command or not Path(
            child_data["execution_output"]
        ).is_relative_to(Path(child_data["execution_root"]) / "results/runs"):
            raise ArithmeticError("actual complete child command/output changed")
        authenticate_source(root, child_data["source"])
        expected_raw = {"observer.json", "native.json", "native-events.jsonl"} | {
            f"read-{i}{suffix}"
            for i in range(3)
            for suffix in (".json", "-events.jsonl")
        }
        if set(child_data["raw_sha256"]) != expected_raw or any(
            p.is_symlink() for p in folder.rglob("*")
        ):
            raise ArithmeticError(
                "complete explicit raw observation inventory required"
            )
        actual_raw = {
            str(p.relative_to(folder)) for p in folder.rglob("*") if p.is_file()
        }
        if actual_raw != expected_raw | {
            "child.json",
            "command.log",
            "measurement.json",
        }:
            raise ArithmeticError("unexpected or missing raw diagnostic member")
        for name, checksum in child_data["raw_sha256"].items():
            path = folder / name
            if (
                path.is_symlink()
                or not path.resolve().is_relative_to(folder.resolve())
                or digest(path) != checksum
            ):
                raise ArithmeticError("saved raw observation changed")
        observed = json.loads((folder / "observer.json").read_text())
        results = validate_observed(
            observed,
            enabled=mode == "reuse",
            pid=measurement["child_pid"],
            thread=child_data["thread"],
            config=c,
        )
        if (
            results != child_data["summaries"]
            or observed["reads"] != child_data["read_clocks"]
            or observed["state"]["rows"][1]["collection_sha256"]
            != child_data["context_identity_sha256"]
            or observed["replays"] != child_data["identity"]["computed_replays"]
        ):
            raise ArithmeticError("saved computed summary changed")
        for row in observed["reads"]:
            if any(
                not start[k] <= row["before"][k] <= row["after"][k] <= end[k]
                for k in ("cpu", "wall")
            ):
                raise ArithmeticError("complete read outside actual parent body")
        for index, row in enumerate(observed["reads"]):
            stream = folder / f"read-{index}-events.jsonl"
            if [json.loads(line) for line in stream.read_bytes().splitlines()] != row[
                "events"
            ] or json.loads((folder / f"read-{index}.json").read_text()) != row:
                raise ArithmeticError(
                    "independently retained boundary/event stream changed"
                )
        native = json.loads((folder / "native.json").read_text())
        if native["implementation_config"] != dict(
            instrumentation_config=CONFIG,
            instrumentation_sha256=CONFIG_SHA,
            admission_evidence=False,
        ):
            raise ArithmeticError("saved instrumentation must reject base admission")
        native["implementation_config"] = native["base_implementation_config"]
        check_execution(root, native)
        if (
            native["normalization"]["pid"] != measurement["child_pid"]
            or native["normalization"]["thread"] != child_data["thread"]
        ):
            raise ArithmeticError("base parent and observer producer differ")
        own = json.loads((root / c["base_own"]).read_text())
        if validate_rows(native, own) != native["verification"]:
            raise ArithmeticError("saved scalar comparison verification changed")
        gathered = gather_independent(native["native_batch"], own)
        if any(native[k] != v for k, v in gathered.items()):
            raise ArithmeticError("complete native workers and scalar rows differ")
        validate_journal(
            folder / native["journal"]["file"],
            native["journal"],
            native["source_sha256"],
            {
                3: native["events"],
                **{w["slot"]: w["messages"] for w in native["native_batch"]["workers"]},
            },
        )
        clocks = [
            dict(
                slot=w["slot"], wall_seconds=m["seconds"], cpu_seconds=m["cpu_seconds"]
            )
            for w in native["native_batch"]["workers"]
            for m in w["messages"]
            if m.get("kind") == "done"
        ]
        if clocks != child_data["worker_body_clocks"] or any(
            type(x[k]) not in (int, float) or not math.isfinite(x[k]) or x[k] < 0
            for x in clocks
            for k in ("wall_seconds", "cpu_seconds")
        ):
            raise ArithmeticError("saved separate worker clocks changed")
        if any(w["wall_seconds"] > end["wall"] - start["wall"] for w in clocks):
            raise ArithmeticError(
                "individual worker body exceeds enclosing parent body"
            )
        if (
            previous.numerical_identity(native, observed["replays"])
            != child_data["identity"]
        ):
            raise ArithmeticError("complete raw numerical identity changed")
        children.append(child_data)
        ledgers.append(results)
    try:
        pair_identity(children)
    except ArithmeticError:
        pass
    else:
        raise ArithmeticError("original pair gate unexpectedly accepted rejected data")
    if (
        type(pair_data["seconds"]) not in (int, float)
        or not 0 < pair_data["seconds"] < 1800
    ):
        raise ArithmeticError("saved pair identity/shared clock changed")
    if (
        pair_data["measurements"][0]["started_perf_counter"]
        + pair_data["measurements"][0]["seconds"]
        > pair_data["measurements"][1]["started_perf_counter"]
    ):
        raise ArithmeticError("pair child clocks overlap")
    span = (
        pair_data["measurements"][-1]["started_perf_counter"]
        + pair_data["measurements"][-1]["seconds"]
        - pair_data["measurements"][0]["started_perf_counter"]
    )
    if pair_data["seconds"] < span:
        raise ArithmeticError("shared pair clock omits child execution")
    return dict(pair=pair_data, children=children, ledgers=ledgers)


def main() -> None:
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--root',type=Path,required=True)
    parser.add_argument('--output',type=Path,required=True)
    args=parser.parse_args();root=args.root.resolve();destination=args.output.resolve()
    if destination.exists():raise FileExistsError('retain every prior diagnosis report')
    source=root/'scripts/compare_context_reuse.py'
    assert hashlib.sha256(source.read_bytes()).hexdigest()==SOURCE_SHA
    sys.path[:0]=[str(root/'src'),str(root/'scripts')]
    import compare_context_reuse as runner
    from compact_suit.admission import published
    from poker_ai.evaluation.compact_opponent_guard import isolated
    global CONFIG,CONFIG_SHA,ENTRY,canonical,digest,contract,pair_identity,previous,validate_observed
    CONFIG,CONFIG_SHA,ENTRY=runner.CONFIG,runner.CONFIG_SHA,runner.ENTRY
    canonical,digest,contract,pair_identity,previous=runner.canonical,runner.digest,runner.contract,runner.pair_identity,runner.previous
    validate_observed=runner.validate_observed
    folder=root/runner.FOLDER/'pair';manifest_path=folder.with_name('pair-raw-manifest.json');manifest=json.loads(manifest_path.read_text())
    assert manifest['status']=='failed_retained' and manifest['stage']=='pair' and len(manifest['files'])==27
    for row in manifest['files']:
        p=root/row['file'];assert not p.is_symlink() and p.resolve().is_relative_to(root)
        assert len(p.read_bytes())==row['bytes'] and digest(p)==row['sha256'];published(root,p)
    published(root,manifest_path)
    before={str(p.relative_to(folder)):digest(p) for p in folder.rglob('*') if p.is_file()}
    # The exact previously tested guard list is archived with source qualification.
    guard=root/runner.FOLDER/'checks/report_execution_guard.py.txt'
    tree=ast.parse(guard.read_text());blocked=ast.literal_eval(next(n.value for n in tree.body if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='blocked' for t in n.targets)))
    assert len(blocked)==103
    attempts=[]
    def forbidden(*a: Any,**k: Any) -> Any:
        attempts.append('forbidden');raise AssertionError('failure report attempted numerical work')
    with isolated(worker=False),ExitStack() as stack:
        for name in blocked:pkgutil.resolve_name(name)
        descriptors={'poker_ai.postflop.matrix.MatrixTree.build','poker_ai.postflop.tree.SpotTree.build','grouped_depth_support.BASE_DESCRIPTOR'}
        for name in blocked:stack.enter_context(patch(name,classmethod(forbidden) if name in descriptors else forbidden))
        data=read_rejected_pair(root,folder)
        try:runner.saved_pair(root,folder)
        except ArithmeticError:pass
        else:raise AssertionError('original valid-pair parser stopped rejecting')
        a,b=data['children'];x,y=copy.deepcopy(a['identity']),copy.deepcopy(b['identity']);differences=[]
        for category,key in WORK_PATHS:
            left,right=x[category].pop(key),y[category].pop(key)
            assert type(left) is int and type(right) is int and 0<=right<left
            differences.append(dict(path=category+'.'+key,ordinary=left,reuse=right,kind='work_counter',original_gate_rejected=True))
        assert canonical(x)==canonical(y)
        assert a['context_identity_sha256']==b['context_identity_sha256']
        assert differences[0]['ordinary']-differences[0]['reuse']==differences[1]['ordinary']-differences[1]['reuse']
        assert differences[2]['ordinary']-differences[2]['reuse']==differences[3]['ordinary']-differences[3]['reuse']
        reads,commands=runner.exports(data)
    assert not attempts and before=={str(p.relative_to(folder)):digest(p) for p in folder.rglob('*') if p.is_file()}
    outer=json.loads(folder.with_name('pair-outer.json').read_text());pair=data['pair']
    assert outer['status']=='failed_retained' and outer['returncode']!=0 and outer['cap_seconds']==1800 and outer['interrupted'] is None
    assert digest(folder.with_name('pair-outer.log'))==outer['command_log_sha256']
    assert 0<outer['seconds']<1800 and pair['seconds']<=outer['seconds']
    for m in pair['measurements']:
        assert outer['started_perf_counter']<=m['started_perf_counter']<=m['started_perf_counter']+m['seconds']<=outer['started_perf_counter']+outer['seconds']
    destination.mkdir(parents=True)
    for name,rows in [('reads.csv',reads),('commands.csv',commands),('differences.csv',differences)]:
        with (destination/name).open('x',newline='') as stream:
            writer=csv.DictWriter(stream,fieldnames=list(rows[0]));writer.writeheader();writer.writerows(rows)
    summary=dict(status='rejected_pair_post_outcome_diagnosis',original_gate_rejected=True,valid_pair=False,admission_evidence=False,scientific_source_sha256=SOURCE_SHA,raw_manifest_sha256=digest(manifest_path),source_fingerprint=a['source']['fingerprint'],context_witness_sha256=a['context_identity_sha256'],full_identity_sha256=[hashlib.sha256(canonical(c['identity'])).hexdigest() for c in data['children']],post_hoc_nonwork_projection_sha256=hashlib.sha256(canonical(x)).hexdigest(),only_differences=WORK_PATHS,work_differences=differences,reads=reads,commands=commands,outer_seconds=outer['seconds'],blocked=blocked,forbidden_attempts=attempts,raw_unchanged=True,limitations='The original preregistered complete identity gate FAILED and remains failed. Removing four work counters here is a POST-OUTCOME diagnosis, never prospective admission or a validated speedup. All remaining saved result fields and context witnesses match. One fixed ordered known-game comparison; iterator clocks include consumer work; parallel worker CPU is not elapsed time. No new training, production forecast/default/strength admission. No retry or source/gate modification.')
    (destination/'summary.json').write_bytes(canonical(summary)+b'\n')
    matplotlib=importlib.import_module('matplotlib');matplotlib.use('Agg');plt=importlib.import_module('matplotlib.pyplot')
    fig,axes=plt.subplots(1,3,figsize=(12,4))
    for axis,key,title in zip(axes,['complete_wall_seconds','parent_body_cpu_seconds','sum_worker_body_cpu_seconds'],['Complete elapsed time','Parent body CPU','Sum of worker body CPU'],strict=True):
        axis.bar([r['mode'] for r in commands],[r[key] for r in commands]);axis.set_title(title);axis.set_ylabel('Seconds')
    fig.suptitle('REJECTED pair: diagnostic clocks; no validated speedup or admission');fig.tight_layout();fig.savefig(destination/'rejected-clocks.png',dpi=150);plt.close(fig)
    fig,axes=plt.subplots(1,2,figsize=(12,5))
    colors=['#0072B2','#E69F00','#009E73','#CC79A7']
    for index,axis in enumerate(axes,1):
        chosen=[r for r in reads if r['read']==index];bottom=[0.,0.]
        for color,key,label in zip(colors,['sampled_dictionary_cpu_seconds','full_dictionary_cpu_seconds','prior_iterator_inclusive_cpu_seconds','other_archive_cpu_seconds'],['First dictionary','Second dictionary','Prior iterator + consumer','Other archive work'],strict=True):
            vals=[r[key] for r in chosen];axis.bar([r['mode'] for r in chosen],vals,bottom=bottom,color=color,label=label);bottom=[a+b for a,b in zip(bottom,vals,strict=True)]
        axis.set_ylabel('Parent CPU seconds inside archive check');axis.set_title('Before reference' if index==1 else 'After reference')
    axes[1].legend(loc='upper left',bbox_to_anchor=(1.02,1));fig.suptitle('REJECTED pair: request work, including copy/check costs');fig.tight_layout();fig.savefig(destination/'rejected-context-work.png',dpi=150);plt.close(fig)
    assert before=={str(p.relative_to(folder)):digest(p) for p in folder.rglob('*') if p.is_file()}
    print(json.dumps(dict(status=summary['status'],valid_pair=False,forbidden_calls=len(attempts),exports=[n for n in ('reads.csv','commands.csv','differences.csv','summary.json')],commands=commands,differences=differences)))


if __name__=='__main__':main()
