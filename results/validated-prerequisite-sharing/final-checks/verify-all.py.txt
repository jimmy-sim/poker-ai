"""Verify retained final local and hosted checks for the data-only report."""

import hashlib
import json
import subprocess
from pathlib import Path

folder = Path(__file__).resolve().parent
root = folder.parents[2]
manifest = json.loads((folder / "manifest.json").read_bytes())
assert manifest["status"] == "final_failed_experiment_report_qualified"
head = manifest["checked_commit"]
assert head == "32d81ed838b8c5755405ca8892bdbf2cc751198f"
assert subprocess.check_output(["git", "rev-parse", f"{head}^{{tree}}"], cwd=root, text=True).strip() == manifest["source_tree"]
assert hashlib.sha256((root / "uv.lock").read_bytes()).hexdigest() == manifest["lock_sha256"]
report = (root / "results/validated-prerequisite-sharing/report.json").read_bytes()
assert hashlib.sha256(report).hexdigest() == manifest["report_sha256"]
assert len(manifest["files"]) == len({row["file"] for row in manifest["files"]})
for row in manifest["files"]:
    raw = (folder / row["file"]).read_bytes()
    assert len(raw) == row["bytes"]
    assert hashlib.sha256(raw).hexdigest() == row["sha256"]
local = json.loads((folder / "local/verified.json").read_bytes())
hosted = json.loads((folder / "hosted/verified.json").read_bytes())
assert local["status"] == "four_clean_final_local_roles_and_hooks_passed"
assert local["checked_commit"] == head
assert local["base_tests_each"] == 5605 and local["optional_tests_each"] == 2510
assert local["report_sha256"] == manifest["report_sha256"]
assert hosted["status"] == "both_exact_hosted_jobs_pass"
assert hosted["head"] == head and hosted["source_tree"] == manifest["source_tree"]
assert hosted["whole_tree_equal"] is True
assert {row["role"] for row in hosted["rows"]} == {"hosted311", "hosted312"}
print(json.dumps({"status": "final_report_checks_verified", "checked_commit": head, "members": len(manifest["files"])}, indent=2))
