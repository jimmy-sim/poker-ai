"""Seal all six fixed candidate profile paths before separate admission."""

import hashlib
import json
import shutil
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
RAW = Path(__file__).resolve().parent
OUT = ROOT / "results/known-turn-regret-recency/profile"
SOURCE = "133a5a77c19431bf1c220bc89d000c6a146cf53161532bcf26eab0499aa2cf10"


def main() -> None:
    if OUT.exists() or OUT.is_symlink():
        raise FileExistsError("retain earlier profile")
    sys.path.insert(0, str(ROOT / "scripts"))
    from known_turn_regret_recency.forecast import forecast
    from known_turn_regret_recency.gate import PROOF, SOURCE as SOURCE_STAGE
    from known_turn_regret_recency.gate import clean_head, require_archive, source_fingerprint
    from known_turn_regret_recency.plan import context, contract, paths
    from study_known_turn_regret_recency import verify_work
    from turn_exact_transfer.capture import replay_path

    head = clean_head(ROOT)
    if source_fingerprint(ROOT) != SOURCE:
        raise ArithmeticError("tested source changed")
    for folder, status in ((SOURCE_STAGE, "source_qualified"), (PROOF, "proof_passed")):
        require_archive(ROOT, ROOT / folder, status, SOURCE)
    state = json.loads((RAW / "job-state.json").read_bytes())
    result = json.loads((RAW / "raw/result.json").read_bytes())
    started = json.loads((RAW / "raw/started.json").read_bytes())
    saved_forecast = json.loads((RAW / "raw/forecast.json").read_bytes())
    if (
        state["status"] != "passed"
        or state["exit_code"] != 0
        or state["cap_seconds"] != 900
        or not 0 < state["observed_wall_seconds"] < 900
        or state["source_head"] != head
        or state["source_fingerprint"] != SOURCE
        or result["status"] != "completed"
        or result["head"] != head
        or result["source_fingerprint"] != SOURCE
        or result["stage"] != "profile"
        or started["stage"] != "profile"
        or started["source_fingerprint"] != SOURCE
        or saved_forecast != forecast(result)
        or (RAW / "launch.exit").read_text().strip() != "0"
    ):
        raise ArithmeticError("fixed capped profile identity or forecast changed")
    plan = contract(ROOT)
    for spec in paths():
        scoring = context(ROOT, spec.game)
        for repeat in range(3):
            folder = RAW / "raw" / f"repeat{repeat}" / f"path{spec.index:02d}"
            summary = replay_path(scoring, folder)
            verify_work(summary, spec, (1_000_000,), plan)
            if len(summary["captures"]) != 2 or summary["recall_violations"] != 0:
                raise ArithmeticError("fixed profile capture or recall changed")
    if any(path.is_symlink() for path in (RAW / "raw").rglob("*")):
        raise ValueError("profile raw symlink")
    shutil.copytree(RAW / "raw", OUT)
    for name in (
        "job-state.json",
        "launch.exit",
        "launch.log",
        "launcher.log",
        "launcher.py",
        "package_stage.py",
        "pr-body.md",
    ):
        source = RAW / name
        target = OUT / (name + ".txt" if name.endswith(".py") else name)
        shutil.copyfile(source, target)
    files = []
    for path in sorted(path for path in OUT.rglob("*") if path.is_file()):
        raw = path.read_bytes()
        files.append(
            dict(
                file=path.relative_to(OUT).as_posix(),
                bytes=len(raw),
                sha256=hashlib.sha256(raw).hexdigest(),
            )
        )
    (OUT / "manifest.json").write_text(
        json.dumps(
            dict(status="profile_published", source_fingerprint=SOURCE, files=files),
            sort_keys=True,
            indent=2,
        )
        + "\n"
    )
    print("profile_published", len(files), OUT)


if __name__ == "__main__":
    main()
