"""Seal separate strict compute admission from published profile bytes."""

import hashlib
import json
import shutil
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
RAW = Path(__file__).resolve().parent
OUT = ROOT / "results/known-turn-regret-recency/admission"
SOURCE = "133a5a77c19431bf1c220bc89d000c6a146cf53161532bcf26eab0499aa2cf10"


def main() -> None:
    if OUT.exists() or OUT.is_symlink():
        raise FileExistsError("retain earlier admission")
    sys.path.insert(0, str(ROOT / "scripts"))
    from known_turn_regret_recency.forecast import forecast
    from known_turn_regret_recency.gate import PROFILE, PROOF, SOURCE as SOURCE_STAGE
    from known_turn_regret_recency.gate import clean_head, require_archive, source_fingerprint

    head = clean_head(ROOT)
    if source_fingerprint(ROOT) != SOURCE:
        raise ArithmeticError("tested source changed")
    for folder, status in (
        (SOURCE_STAGE, "source_qualified"),
        (PROOF, "proof_passed"),
        (PROFILE, "profile_published"),
    ):
        require_archive(ROOT, ROOT / folder, status, SOURCE)
    state = json.loads((RAW / "job-state.json").read_bytes())
    decision = json.loads((RAW / "raw/decision.json").read_bytes())
    receipt = json.loads((RAW / "raw/receipt.json").read_bytes())
    profile = json.loads((ROOT / PROFILE / "result.json").read_bytes())
    if (
        state["status"] != "passed"
        or state["exit_code"] != 0
        or state["cap_seconds"] != 180
        or not 0 < state["observed_wall_seconds"] < 180
        or state["source_head"] != head
        or state["source_fingerprint"] != SOURCE
        or decision != forecast(profile)
        or decision["status"] != "admitted"
        or not decision["strict_bound_seconds"] < 7200
        or receipt != dict(
            head=head,
            source_fingerprint=SOURCE,
            profile=PROFILE.as_posix(),
            decision_status="admitted",
        )
        or (RAW / "launch.exit").read_text().strip() != "0"
    ):
        raise ArithmeticError("separate strict admission changed")
    shutil.copytree(RAW / "raw", OUT)
    for name in (
        "job-state.json",
        "launch.exit",
        "launch.log",
        "launcher.log",
        "launcher.py",
        "package_stage.py",
    ):
        source = RAW / name
        target = OUT / (name + ".txt" if name.endswith(".py") else name)
        shutil.copyfile(source, target)
    files = []
    for path in sorted(path for path in OUT.rglob("*") if path.is_file()):
        raw = path.read_bytes()
        files.append(
            dict(
                file=path.relative_to(OUT).as_posix(),
                bytes=len(raw),
                sha256=hashlib.sha256(raw).hexdigest(),
            )
        )
    (OUT / "manifest.json").write_text(
        json.dumps(dict(status="admitted", source_fingerprint=SOURCE, files=files), sort_keys=True, indent=2)
        + "\n"
    )
    print("admitted", len(files), OUT)


if __name__ == "__main__":
    main()
