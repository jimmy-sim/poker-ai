"""Seal both complete own candidate paths before any reference comparison."""

import hashlib
import json
import shutil
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
RAW = Path(__file__).resolve().parent
OUT = ROOT / "results/known-turn-regret-recency/own"
SOURCE = "133a5a77c19431bf1c220bc89d000c6a146cf53161532bcf26eab0499aa2cf10"


def main() -> None:
    if OUT.exists() or OUT.is_symlink():
        raise FileExistsError("retain earlier own study")
    sys.path.insert(0, str(ROOT / "scripts"))
    from known_turn_regret_recency.gate import SOURCE as SOURCE_STAGE
    from known_turn_regret_recency.gate import PROOF, PROFILE, ADMISSION
    from known_turn_regret_recency.gate import clean_head, require_archive, require_admission, source_fingerprint
    from known_turn_regret_recency.plan import LABELS, context, contract, paths
    from study_known_turn_regret_recency import verify_work
    from turn_exact_transfer.capture import replay_path

    head = clean_head(ROOT)
    if source_fingerprint(ROOT) != SOURCE:
        raise ArithmeticError("tested own source changed")
    for folder, status in (
        (SOURCE_STAGE, "source_qualified"),
        (PROOF, "proof_passed"),
        (PROFILE, "profile_published"),
        (ADMISSION, "admitted"),
    ):
        require_archive(ROOT, ROOT / folder, status, SOURCE)
    require_admission(ROOT, SOURCE)
    state = json.loads((RAW / "job-state.json").read_bytes())
    result = json.loads((RAW / "raw/result.json").read_bytes())
    started = json.loads((RAW / "raw/started.json").read_bytes())
    if (
        state["status"] != "passed"
        or state["exit_code"] != 0
        or state["cap_seconds"] != 7200
        or not 0 < state["observed_wall_seconds"] < 7200
        or state["source_head"] != head
        or state["source_fingerprint"] != SOURCE
        or result["status"] != "completed"
        or result["stage"] != "study"
        or result["head"] != head
        or result["source_fingerprint"] != SOURCE
        or result["assignments"] != [[0], [1]]
        or len(result["workers"]) != 2
        or started["stage"] != "study"
        or started["source_fingerprint"] != SOURCE
        or (RAW / "launch.exit").read_text().strip() != "0"
    ):
        raise ArithmeticError("fixed own study identity or cap changed")
    plan = contract(ROOT)
    for spec in paths():
        scoring = context(ROOT, spec.game)
        summary = replay_path(scoring, RAW / "raw/study" / f"path{spec.index:02d}")
        verify_work(summary, spec, LABELS, plan)
        if len(summary["captures"]) != 6 or summary["recall_violations"] != 0:
            raise ArithmeticError("fixed own capture or recall changed")
        for capture in summary["captures"]:
            score = capture["score"]["metrics"]["exact_exploitability_chips"]
            if not 0 <= score < float("inf"):
                raise ArithmeticError("nonfinite own exact score")
            if capture["metadata"]["readout"] == "linear_average":
                support = capture["support"]
                if support is None or support["whole"]["all"]["uniform_reach_fraction"] < 0:
                    raise ArithmeticError("own average reach missing")
    if any(path.is_symlink() for path in (RAW / "raw").rglob("*")):
        raise ValueError("own raw symlink")
    shutil.copytree(RAW / "raw", OUT)
    for name in (
        "job-state.json",
        "launch.exit",
        "launch.log",
        "launcher.log",
        "launcher.py",
        "package_stage.py",
        "pr-body.md",
    ):
        source = RAW / name
        target = OUT / (name + ".txt" if name.endswith(".py") else name)
        shutil.copyfile(source, target)
    files = []
    for path in sorted(path for path in OUT.rglob("*") if path.is_file()):
        raw = path.read_bytes()
        files.append(
            dict(
                file=path.relative_to(OUT).as_posix(),
                bytes=len(raw),
                sha256=hashlib.sha256(raw).hexdigest(),
            )
        )
    (OUT / "manifest.json").write_text(
        json.dumps(dict(status="own_published", source_fingerprint=SOURCE, files=files), sort_keys=True, indent=2)
        + "\n"
    )
    print("own_published", len(files), OUT)


if __name__ == "__main__":
    main()
