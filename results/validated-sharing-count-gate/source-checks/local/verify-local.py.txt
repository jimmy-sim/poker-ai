"""Verify exact-head base, optional, hooks and source guard receipts."""
import hashlib
import json
import re
import subprocess
from pathlib import Path
root=Path.cwd()
folder=Path(__file__).resolve().parent
head='182fd7c44f3bccf2a987ec304a04a05219743e32'
fingerprint='b475bfefb92963f25458d97260264743ff4d751b4513c307529742eaf10fc97d'
lock='db5529861ae56225daaf8a18513013e52bee88ef667227855b719a7e1364250c'
assert subprocess.run(['git','merge-base','--is-ancestor',head,'HEAD'],cwd=root,check=False).returncode==0
assert hashlib.sha256((root/'uv.lock').read_bytes()).hexdigest()==lock
suite=json.loads((folder/'optional-suites.json').read_bytes())
assert len(suite)==72 and all((root/path).is_file() for path in suite)
for minor,version,numpy in [('311','3.11.17','2.4.6'),('312','3.12.15','2.5.4')]:
 base=(folder/f'base{minor}.log').read_text()
 assert (folder/f'base{minor}.exit').read_text().strip()=='0'
 assert 'All checks passed!' in base and '1015 files already formatted' in base
 assert 'Success: no issues found in 848 source files' in base
 assert re.search(r'5629 passed, 72 skipped, 150 deselected in [0-9.]+s',base)
 for side in ('pre','post'):
  guard=json.loads((folder/f'base{minor}-{side}.json').read_bytes())
  assert guard['head']==head and guard['dirty'] is False
  assert (guard['python'],guard['numpy'])==(version,numpy)
  assert guard['source_fingerprint']==fingerprint and guard['source_files']==458
  assert guard['lock_sha256']==lock and guard['launcher_pythonpath_absent'] is True
  assert guard['import_root'].endswith('/src/poker_ai/__init__.py')
 optional=(folder/f'optional{minor}.log').read_text()
 expected='2535 passed in ' if minor=='311' else '2534 passed, 1 skipped in '
 assert expected in optional
 for field,value in [('ACTUAL_PYTHON',version),('ACTUAL_NUMPY',numpy),('CHECKED_COMMIT',head),('SOURCE_FINGERPRINT',fingerprint)]:
  assert optional.count(f'{field}={value}')==2
 receipt=json.loads((folder/f'optional{minor}.json').read_bytes())
 assert receipt['status']=='passed' and receipt['role']=='optional'+minor
 assert receipt['checked_commit']==head and receipt['suite_count']==72
 assert (receipt['python'],receipt['numpy'])==(version,numpy)
 assert receipt['source_fingerprint']==fingerprint and receipt['no_launcher_PYTHONPATH'] is True
 license=json.loads((folder/f'optional{minor}-license.json').read_bytes())
 assert license['python']==version and license['license']=='Apache-2.0'
 assert license['bytes']==11358 and license['sha256']=='cfc7749b96f63bd31c3c42b5c471bf756814053e847c10f3eb003417bc523d30'
assert (folder/'hooks.exit').read_text().strip()=='0'
assert (folder/'hooks.log').read_text().count('Passed')==2
for path in ('scripts/validated_prerequisite_sharing_support.py','scripts/prove_validated_prerequisite_sharing.py','scripts/compare_validated_prerequisite_sharing.py','scripts/launch_validated_sharing_optional.py'):
 assert subprocess.check_output(['git','show',f'{head}:{path}'],cwd=root)==(root/path).read_bytes()
print('four exact-head local roles, hooks, source bytes, license and guards verified')
