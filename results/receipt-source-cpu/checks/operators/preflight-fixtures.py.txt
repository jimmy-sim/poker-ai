 def check():
  if kind=='rejected':raise ArithmeticError('own publication not verified')
  if kind=='interrupt':raise KeyboardInterrupt()

 result=m.run([sys.executable,'-c','from pathlib import Path;import sys;Path(sys.argv[1]).write_text("ran")',str(marker)],destination,.01 if kind=='exhausted' else 10,preflight=check)
 expected=0 if kind=='pass' else 126 if kind=='rejected' else 130 if kind=='interrupt' else 124
