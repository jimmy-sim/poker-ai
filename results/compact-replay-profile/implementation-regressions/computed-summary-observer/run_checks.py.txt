"""Final immutable-checkout base and focused all-extras publication checks."""

import json
import os
import subprocess
import sys
import time
from pathlib import Path

work = Path(__file__).resolve().parent
root = work / "compact-replay-profile-checkout"
folder = work / "compact-replay-profile-history-checks"
folder.mkdir(exist_ok=True)
name = sys.argv[1]
assert name in ("base311", "base312", "optional311", "optional312")
optional = name.startswith("optional")
minor = name[-3:]
version = "3.11.17" if minor == "311" else "3.12.15"
python = work / "python" / f"cpython-{version}-macos-aarch64-none/bin/python3.{minor[-2:]}"
env = dict(os.environ, UV_PROJECT_ENVIRONMENT=str(work / f"compact-replay-profile-{name}"), UV_PYTHON=str(python))
uv = ["uv", "run", "--locked", *(["--all-extras"] if optional else [])]
started = time.perf_counter()
with (folder / f"{name}-install.log").open("x") as stream:
    subprocess.run(["uv", "sync", "--locked", "--dev", *(["--all-extras"] if optional else [])], cwd=root, env=env, stdout=stream, stderr=subprocess.STDOUT, check=True)
guard = """import pathlib,subprocess,sys,numpy,poker_ai
import importlib.util
spec=importlib.util.spec_from_file_location("profile_tool",pathlib.Path.cwd()/"scripts/profile_compact_replay.py")
module=importlib.util.module_from_spec(spec);spec.loader.exec_module(module)
def source(root,clean=True):return module.preflight(root)[1]
root=pathlib.Path.cwd()
assert subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()=='567a9b1345df8b7dcd434cb70787ca2acdcac401'
assert pathlib.Path(poker_ai.__file__).resolve().is_relative_to(root/'src')
assert sys.version.split()[0]==VERSION
assert numpy.__version__==NUMPY
s=source(root,clean=True)
assert s['dirty'] is False
assert s['fingerprint']=='cd4d6962702156153fdb914b5fd370c3f9773835aa86949fccfdbfee5a749569'
print({'python':sys.version.split()[0],'numpy':numpy.__version__,'module':poker_ai.__file__,'source_sha256':s['fingerprint']})
""".replace("VERSION", repr(version)).replace("NUMPY", repr("2.4.6" if minor == "311" else "2.5.4"))
commands = [uv + ["python", "-c", guard]]
if optional:
    suites = ["tests/test_compact_replay_profile.py","tests/test_batch_cost_stats.py","tests/test_compact_opponent_isolation.py","tests/test_compact_opponent_admission.py","tests/test_compact_opponent_clock.py"]
    commands.append(uv + ["pytest","-q",*suites])
else:
    commands.extend((uv + ["ruff", "check", "."], uv + ["ruff", "format", "--check", "."], uv + ["mypy", "--strict", "src", "tests", "scripts"], uv + ["pytest", "-q", "-m", "not slow"], uv + ["pre-commit", "validate-config"], uv + ["pre-commit", "run", "--all-files"]))
commands.append(uv + ["python", "-c", guard])
began = time.perf_counter()
outcome = "failed"
try:
    with (folder / f"{name}.log").open("x") as stream:
        for command in commands:
            stream.write(json.dumps(dict(command=command)) + "\n")
            stream.flush()
            subprocess.run(command, cwd=root, env=env, stdout=stream, stderr=subprocess.STDOUT, check=True)
    outcome = "passed"
finally:
    (folder / f"{name}-receipt.json").write_text(json.dumps(dict(name=name, status=outcome, checked_commit="567a9b1345df8b7dcd434cb70787ca2acdcac401", scientific_source_sha256="cd4d6962702156153fdb914b5fd370c3f9773835aa86949fccfdbfee5a749569", scope="complete standalone replay profiler/identity/caps/failure/viewer and unchanged raw-profiler/constructor-isolation/admission/clock checks, all extras; actual complete432 tiny replay. No formal poker profile or training" if optional else "complete base lint/format/types/non-slow/tests/hooks", check_seconds=time.perf_counter()-began, setup_and_check_seconds=time.perf_counter()-started), indent=2)+"\n")
print(name + ": complete, source/runtime guarded")
