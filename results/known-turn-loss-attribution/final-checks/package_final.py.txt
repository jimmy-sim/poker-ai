"""Seal exact-report qualification, including canceled intermediate workflows."""
import gzip
import hashlib
import json
import shutil
import subprocess
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
RAW = Path(__file__).resolve().parent
CANCELED = ROOT / "results/runs/known-turn-loss-attribution-publication-ci-20261124"
OUT = ROOT / "results/known-turn-loss-attribution/final-checks"
HEAD = "5d9f972b191f9c34cbbd1c8b0c3942c9edd265ec"
SOURCE = "11125bcb2b866b409d9e3bfd4d7f51c3c6fde53c1fc80575c14d8f526516ed68"
REPORT_SHA = "3f0989918a80bad24884dcd9ea9c745851d4fd4e74237cf61952270cd68362c1"
ROLES = ("base311", "base312", "optional311", "optional312")


def copy(src: Path, label: str) -> None:
    if not src.is_file():
        raise FileNotFoundError(src)
    target = OUT / label
    if src.suffix == ".log":
        with target.with_name(target.name + ".gz").open("xb") as stream:
            with gzip.GzipFile(filename="", mode="wb", fileobj=stream, mtime=0) as zipped:
                zipped.write(src.read_bytes())
    elif src.suffix in (".py", ".zsh"):
        shutil.copyfile(src, target.with_name(target.name + ".txt"))
    else:
        shutil.copyfile(src, target)


def main() -> None:
    if OUT.exists() or OUT.is_symlink():
        raise FileExistsError("retain earlier final receipt")
    head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip()
    dirty = subprocess.check_output(["git", "status", "--porcelain=v1"], cwd=ROOT)
    if head != HEAD or dirty:
        raise ArithmeticError("tested report changed or dirty")
    if hashlib.sha256((ROOT / "docs/known-turn-loss-attribution-result.md").read_bytes()).hexdigest() != REPORT_SHA:
        raise ArithmeticError("exact report bytes changed")
    state = json.loads((RAW / "driver-state.json").read_bytes())
    receipts = [json.loads((RAW / f"{role}.json").read_bytes()) for role in ROLES]
    hosted = json.loads((RAW / "hosted.json").read_bytes())
    if (
        state["status"] != "passed" or state["head"] != HEAD
        or state["completed"] != [{"role": role, "exit_code": 0} for role in ROLES]
        or any(row["status"] != "passed" or row["head"] != HEAD or row["source_fingerprint"] != SOURCE for row in receipts)
        or any(row["suite_count"] != 82 for row in receipts[2:])
        or hosted["headSha"] != HEAD or hosted["conclusion"] != "success"
        or {row["name"]: row["conclusion"] for row in hosted["jobs"]} != {"Python 3.11": "success", "Python 3.12": "success"}
        or (RAW / "exact-hooks.exit").read_text().strip() != "0"
    ):
        raise ArithmeticError("final exact-report qualification incomplete")
    for role, phrase in (("base311", "5832 passed, 73 skipped, 150 deselected"), ("base312", "5832 passed, 73 skipped, 150 deselected"), ("optional311", "2601 passed"), ("optional312", "2599 passed, 2 skipped")):
        if phrase not in (RAW / f"{role}.log").read_text():
            raise ArithmeticError("final test summary missing: " + role)
    canceled = []
    for label, expected in (("source-receipt", "9dc61036"), ("proof-receipt", "5a1e1491"), ("admission-receipt", "b5973bbb"), ("raw-receipt", "577a6b78")):
        row = json.loads((CANCELED / f"{label}.json").read_bytes())
        if not row["headSha"].startswith(expected) or row["conclusion"] != "cancelled":
            raise ArithmeticError("intermediate canceled CI binding changed: " + label)
        canceled.append({"label": label, "run": row["databaseId"], "head": row["headSha"], "conclusion": row["conclusion"]})
    OUT.mkdir(parents=True)
    for name in ("driver-state.json", "driver.py", "runner.py", "launch.zsh", "optional-suites.json", "package_final.py", "report_audit.py", "exact-hooks.exit", "exact-hooks.log", "exact-hooks-first.log", "iteration-failure.md", "hosted.json", "hosted.log", "report-audit.log"):
        copy(RAW / name, name)
    for role in ROLES:
        for suffix in ("json", "log"):
            name = f"{role}.{suffix}"
            copy(RAW / name, name)
    for label in ("source-receipt", "proof-receipt", "admission-receipt", "raw-receipt"):
        for suffix in ("json", "log"):
            copy(CANCELED / f"{label}.{suffix}", f"{label}.{suffix}")
    (OUT / "result.json").write_text(json.dumps({"status": "final_qualified", "tested_head": HEAD, "source_fingerprint": SOURCE, "report_sha256": REPORT_SHA, "scientific_result": "inconclusive_due_binding", "dry_candidate_traced": False, "no_scientific_retry": True, "local_roles": receipts, "hosted_run": hosted["databaseId"], "hosted_jobs": [{"name": j["name"], "conclusion": j["conclusion"]} for j in hosted["jobs"]], "intermediate_canceled_ci_retained": canceled, "hooks": "passed"}, sort_keys=True, indent=2) + "\n")
    files = [{"file": path.name, "bytes": len(path.read_bytes()), "sha256": hashlib.sha256(path.read_bytes()).hexdigest()} for path in sorted(OUT.iterdir())]
    (OUT / "manifest.json").write_text(json.dumps({"status": "final_qualified", "source_fingerprint": SOURCE, "files": files}, sort_keys=True, indent=2) + "\n")
    print("final_qualified", len(files), OUT)

if __name__ == "__main__":
    main()
