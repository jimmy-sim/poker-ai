"""Source/runtime-bound full base and relevant optional diagnostic checks."""
import json
import os
import subprocess
import sys
import time
from pathlib import Path

work = Path(__file__).resolve().parent
root = work / "river-range-checkout"
folder = work / "river-range-checks"
folder.mkdir(exist_ok=True)
name = sys.argv[1]
optional = name.startswith("optional")
minor = name[-3:]
version = {"311": "3.11.17", "312": "3.12.15"}[minor]
python = work / "python" / f"cpython-{version}-macos-aarch64-none/bin/python3.{minor[-2:]}"
env = dict(os.environ, UV_PROJECT_ENVIRONMENT=f"/tmp/poker-river-range-check-{name}", UV_PYTHON=str(python))
extra = ["--all-extras"] if optional else []
uv = ["uv", "run", "--locked", *extra]
started = time.perf_counter()
with (folder / f"{name}-install.log").open("x") as log:
    subprocess.run(["uv", "sync", "--locked", "--dev", *extra], cwd=root, env=env, stdout=log, stderr=subprocess.STDOUT, check=True)
guard = """import pathlib,subprocess,sys,numpy,poker_ai
from poker_ai.evaluation.river_range_io import CONFIG,ENTRIES
from poker_ai.validation.five_card import source_provenance
from poker_ai.postflop.learning_artifacts import sha
root=pathlib.Path.cwd()
assert subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()=='42cd59f31324566b3286fc112e496438f5c484f4'
assert pathlib.Path(poker_ai.__file__).resolve().is_relative_to(root/'src')
assert sys.version.split()[0]==VERSION
assert numpy.__version__==NUMPY
source=source_provenance(root,root/CONFIG,entrypoints=ENTRIES)
assert source['dirty'] is False
print({'python':sys.version.split()[0],'numpy':numpy.__version__,'module':poker_ai.__file__,'source_sha256':sha(source['sha256'])})
""".replace("VERSION", repr(version)).replace("NUMPY", repr("2.4.6" if minor == "311" else "2.5.4"))
commands = [uv + ["python", "-c", guard]]
if optional:
    commands.append(uv + ["pytest", "-q", "tests/test_river_range_math.py", "tests/test_river_range_case.py", "tests/test_river_range_io.py"])
else:
    commands.extend((uv + ["ruff", "check", "."], uv + ["ruff", "format", "--check", "."], uv + ["mypy", "--strict", "src", "tests", "scripts"], uv + ["pytest", "-q", "-m", "not slow"], uv + ["pre-commit", "validate-config"], uv + ["pre-commit", "run", "--all-files"]))
commands.append(uv + ["python", "-c", guard])
began = time.perf_counter()
outcome = "failed"
try:
    with (folder / f"{name}.log").open("x") as log:
        for command in commands:
            log.write(json.dumps(dict(command=command)) + "\n")
            log.flush()
            subprocess.run(command, cwd=root, env=env, stdout=log, stderr=subprocess.STDOUT, check=True)
    outcome = "passed"
finally:
    (folder / f"{name}-receipt.json").write_text(json.dumps(dict(name=name, status=outcome, checked_commit="42cd59f31324566b3286fc112e496438f5c484f4", scope="focused new optional tests" if optional else "complete base lint/format/types/non-slow/tests/hooks", check_seconds=time.perf_counter() - began, setup_and_check_seconds=time.perf_counter() - started), indent=2) + "\n")
print(name + ": complete, source/runtime guarded")
