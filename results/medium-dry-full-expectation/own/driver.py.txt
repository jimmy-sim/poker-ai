"""Run one admitted full-expectation own study with 120 seconds reserved for failure saving."""
import json
import os
import signal
import subprocess
import time
from pathlib import Path

RUN = Path(__file__).resolve().parent
ROOT = RUN.parents[2]
HEAD = "15074539d84c10c9253609ce103e65f4b98baba1"
OUTPUT = RUN / "raw"
COMMAND = [
    "uv", "run", "--locked", "python", "scripts/study_medium_dry_full.py",
    "study", "--output", str(OUTPUT),
]
LEARNING_LIMIT_SECONDS = 7080
WHOLE_CAP_SECONDS = 7200


def save(state: dict[str, object]) -> None:
    path = RUN / "job-state.json"
    with path.open("w") as stream:
        json.dump(state, stream, sort_keys=True, indent=2)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())


if subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip() != HEAD:
    raise RuntimeError("wrong published admission head")
if subprocess.check_output(["git", "status", "--porcelain=v1"], cwd=ROOT):
    raise RuntimeError("dirty source")
if OUTPUT.exists() or not os.environ.get("UV_PYTHON") or not os.environ.get("UV_PROJECT_ENVIRONMENT"):
    raise RuntimeError("existing output or missing locked runtime")
start = time.perf_counter()
state: dict[str, object] = {
    "status": "starting", "driver_pid": os.getpid(), "head": HEAD,
    "protocol_commit": "8770e3243356aad82d6cc07fac70320f04cdfc34",
    "source_fingerprint": "a549976f0811872dcc9057e112f40def8af5d2f3e16b81d1a678fdcf8bc62aab",
    "own_only": True, "reference_used": False,
    "command": COMMAND, "learning_limit_seconds": LEARNING_LIMIT_SECONDS,
    "whole_cap_seconds": WHOLE_CAP_SECONDS, "failure_save_reserve_seconds": 120,
    "output": str(OUTPUT), "log": str(RUN / "launch.log"),
    "uv_python": os.environ["UV_PYTHON"],
    "uv_project_environment": os.environ["UV_PROJECT_ENVIRONMENT"],
    "started_unix_seconds": time.time(),
}
save(state)
with (RUN / "launch.log").open("x") as stream:
    child = subprocess.Popen(COMMAND, cwd=ROOT, env=os.environ.copy(),
                             stdout=stream, stderr=subprocess.STDOUT,
                             start_new_session=True)
    state.update(status="running", child_pid=child.pid, process_group=child.pid)
    save(state)
    interrupted = False
    while child.poll() is None:
        elapsed = time.perf_counter() - start
        if elapsed >= LEARNING_LIMIT_SECONDS and not interrupted:
            os.killpg(child.pid, signal.SIGINT)
            interrupted = True
            state.update(status="interrupting_for_save", interrupted_at_seconds=elapsed)
            save(state)
        if elapsed >= WHOLE_CAP_SECONDS:
            os.killpg(child.pid, signal.SIGTERM)
            try:
                child.wait(timeout=5)
            except subprocess.TimeoutExpired:
                os.killpg(child.pid, signal.SIGKILL)
            break
        time.sleep(0.2)
    code = child.wait()
    elapsed = time.perf_counter() - start
    state.update(status="completed" if code == 0 and (OUTPUT / "result.json").is_file()
                 else "failed_retained", exit_code=code, elapsed_seconds=elapsed,
                 child_pid=None)
    save(state)
    if code != 0 or state["status"] != "completed":
        raise SystemExit(code or 1)
