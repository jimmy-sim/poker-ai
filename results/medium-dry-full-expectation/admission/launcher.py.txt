"""One data-only full-expectation admission from published profile bytes."""
import json
import os
import subprocess
import time
from datetime import datetime, timezone
from pathlib import Path

root = Path(__file__).resolve().parents[3]
folder = Path(__file__).resolve().parent
raw = folder / "raw"
head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=root, text=True).strip()
assert head == "b8d595918b140dda013abceaf02d046a90a6c082"
assert not raw.exists() and not (folder / "job-state.json").exists()
command = ["uv", "run", "--locked", "python", "scripts/study_medium_dry_full.py", "admit", "--output", str(raw)]
env = os.environ.copy()
env["UV_PYTHON"] = "3.11.17"
env["UV_PROJECT_ENVIRONMENT"] = "results/runs/guarded-skip-requalification-20261118/envs/base311"
state = {"status": "starting", "launcher_pid": os.getpid(), "command": command, "source_head": head, "source_fingerprint": "a549976f0811872dcc9057e112f40def8af5d2f3e16b81d1a678fdcf8bc62aab", "environment": {k: env[k] for k in ("UV_PYTHON", "UV_PROJECT_ENVIRONMENT")}, "log": str(folder / "launch.log"), "output": str(raw), "started_utc": datetime.now(timezone.utc).isoformat()}
state_path = folder / "job-state.json"
state_path.write_text(json.dumps(state, sort_keys=True, indent=2) + "\n")
started = time.perf_counter()
with (folder / "launch.log").open("wb") as stream:
    process = subprocess.Popen(command, cwd=root, env=env, stdout=stream, stderr=subprocess.STDOUT, start_new_session=True)
    state.update(status="running", child_pid=process.pid)
    state_path.write_text(json.dumps(state, sort_keys=True, indent=2) + "\n")
    code = process.wait(timeout=60)
state.update(status="passed" if code == 0 else "failed_retained", exit_code=code, observed_wall_seconds=time.perf_counter() - started, finished_utc=datetime.now(timezone.utc).isoformat())
state_path.write_text(json.dumps(state, sort_keys=True, indent=2) + "\n")
(folder / "launch.exit").write_text(f"{code}\n")
print(json.dumps(state, sort_keys=True))
raise SystemExit(code)
