def archive(folder,destination):
 raw=io.BytesIO();members=[]
 with tarfile.open(fileobj=raw,mode='w') as stream:
  for p in sorted(folder.rglob('*')):
   assert not p.is_symlink()
   if not p.is_file():continue
   data=p.read_bytes();name=str(p.relative_to(folder));info=tarfile.TarInfo(name);info.size=len(data);info.mode=0o644;stream.addfile(info,io.BytesIO(data));members.append(dict(file=name,sha256=hashlib.sha256(data).hexdigest(),bytes=len(data)))
 destination.write_bytes(gzip.compress(raw.getvalue(),mtime=0))
 with tarfile.open(fileobj=io.BytesIO(gzip.decompress(destination.read_bytes())),mode='r') as stream:
  for row in members:assert hashlib.sha256(stream.extractfile(row['file']).read()).hexdigest()==row['sha256']
 return members
