"""Single retained 7200-second one-time known-turn own-pair launcher."""
import json
import os
import signal
import subprocess
import time
from datetime import datetime, timezone
from pathlib import Path

root = Path(__file__).resolve().parents[3]
folder = Path(__file__).resolve().parent
raw = folder / "raw"
log = folder / "launch.log"
state = folder / "job-state.json"
head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=root, text=True).strip()
assert head == "a5f1bb5d3d8ff029a696b0c9629d3b7fcf29264d"
assert not raw.exists() and not log.exists() and not state.exists()
command = ["uv", "run", "--locked", "python", "scripts/study_known_turn_mechanism.py", "study", "--output", str(raw)]
env = os.environ.copy()
env["UV_PYTHON"] = "3.11.17"
env["UV_PROJECT_ENVIRONMENT"] = "results/runs/guarded-skip-requalification-20261118/envs/base311"
started = time.perf_counter()
receipt = {"status": "starting", "launcher_pid": os.getpid(), "command": command, "source_head": head, "source_fingerprint": "6e09c02673ca596e6082ea8d0a57ba12cd2533acfaf2c777991c4d29f395e0f4", "cap_seconds": 7200, "environment": {k: env[k] for k in ("UV_PYTHON", "UV_PROJECT_ENVIRONMENT")}, "log": str(log), "output": str(raw), "started_utc": datetime.now(timezone.utc).isoformat()}
state.write_text(json.dumps(receipt, sort_keys=True, indent=2) + "\n")
with log.open("wb") as stream:
    process = subprocess.Popen(command, cwd=root, env=env, stdout=stream, stderr=subprocess.STDOUT, start_new_session=True)
    receipt.update(status="running", child_pid=process.pid)
    state.write_text(json.dumps(receipt, sort_keys=True, indent=2) + "\n")
    try:
        code = process.wait(timeout=max(0.01, 7200 - (time.perf_counter() - started)))
    except subprocess.TimeoutExpired:
        os.killpg(process.pid, signal.SIGTERM)
        try:
            process.wait(timeout=5)
        except subprocess.TimeoutExpired:
            os.killpg(process.pid, signal.SIGKILL)
            process.wait()
        code = 124
    receipt.update(status="passed" if code == 0 else "failed_retained", exit_code=code, observed_wall_seconds=time.perf_counter() - started, finished_utc=datetime.now(timezone.utc).isoformat())
    state.write_text(json.dumps(receipt, sort_keys=True, indent=2) + "\n")
    (folder / "launch.exit").write_text(f"{code}\n")
print(json.dumps(receipt, sort_keys=True))
raise SystemExit(code)
